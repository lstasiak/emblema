18class CampaignWorker: 

19 """What the worker process does with a job, apart from the queue that delivered it. 

20 

21 The composition root is built once and held, because the backbone it reads out of the store 

22 is what makes building one expensive. Everything else here is deserialisation: the settings 

23 and the job carry scalars, and this turns them back into the values a process and a use case 

24 take. No decision is made here — a cell already recorded is recognised by the use case, not 

25 by this. 

26 """

59 def run_campaign_cell(self, arguments: Mapping[str, JobArgument]) -> None: 

60 """Run the cell a job names. 

61 

62 Raises: 

63 ValueError: If the job does not name a cell. 

64 InvalidLabelBudgetError: If the budget it names is not one. 

65 CampaignNotFoundError: If the campaign is unknown.

70 @staticmethod 

71 def command_of(arguments: Mapping[str, JobArgument]) -> RunCampaignCellCommand: 

72 """The coordinates a job carries, read back into a command. 

73 

74 Public because reading a message back is what this entry point is for, and what a 

75 queue delivers is the one input nobody in this process wrote. Every coordinate is 

76 required, the budget included: a message that lost it would otherwise be run at a 

77 budget nobody asked for rather than refused. 

78 

79 Raises: 

80 ValueError: If a coordinate is missing or is not what it has to be. 

81 InvalidLabelBudgetError: If the budget is neither a count nor the word for every 

82 window there is. 

83 """ 

84 campaign, candidate = arguments.get("campaign"), arguments.get("candidate") 

85 budget, seed = arguments.get("budget"), arguments.get("seed") 

86 if not isinstance(campaign, str) or not isinstance(candidate, str): 

87 raise ValueError("a cell names its campaign and its candidate") 

88 if not isinstance(budget, str): 

89 raise ValueError("a cell names the budget of labels it learns from") 

90 if not isinstance(seed, int): 

91 raise ValueError("a cell names the seed of its repeat") 

92 return RunCampaignCellCommand( 

93 campaign=CampaignId.parse(campaign), 

94 cell=CampaignCell( 

95 candidate=CandidateRef(candidate), 

96 budget=LabelBudget.parse(budget), 

97 seed=seed, 

98 ), 

99 )
