13class CampaignWorker: 

14 """What a worker process does with a job, apart from the queue that delivered it. 

15 

16 The same for whichever pool the process serves: reading a message back into a command is 

17 what an entry point is for, and which candidates answer it was settled when the process was 

18 assembled. No decision is made here — a cell already recorded is recognised by the use case, 

19 not by this. 

20 

21 It holds the one use case rather than the whole composition, so that what a job can reach is 

22 what a job needs, and so that a test of reading a message back needs nothing built. 

23 """ 

24 

25 def __init__(self, run_campaign_cell: RunCampaignCell) -> None: 

26 self._run = run_campaign_cell

28 def run_campaign_cell(self, arguments: Mapping[str, JobArgument]) -> None: 

29 """Run the cell a job names. 

30 

31 Raises: 

32 ValueError: If the job does not name a cell. 

33 InvalidLabelBudgetError: If the budget it names is not one. 

34 CampaignNotFoundError: If the campaign is unknown. 

35 UnknownCampaignCellError: If the cell is not one of the campaign's grid. 

36 """ 

37 self._run(self.command_of(arguments)) 

38 

39 @staticmethod 

40 def command_of(arguments: Mapping[str, JobArgument]) -> RunCampaignCellCommand: 

41 """The coordinates a job carries, read back into a command. 

42 

43 Public because reading a message back is what this entry point is for, and what a 

44 queue delivers is the one input nobody in this process wrote. Every coordinate is 

45 required, the budget included: a message that lost it would otherwise be run at a 

46 budget nobody asked for rather than refused. 

47 

48 Raises: 

49 ValueError: If a coordinate is missing or is not what it has to be. 

50 InvalidLabelBudgetError: If the budget is neither a count nor the word for every 

51 window there is. 

52 """ 

53 campaign, candidate = arguments.get("campaign"), arguments.get("candidate") 

54 budget, seed = arguments.get("budget"), arguments.get("seed") 

55 if not isinstance(campaign, str) or not isinstance(candidate, str): 

56 raise ValueError("a cell names its campaign and its candidate") 

57 if not isinstance(budget, str): 

58 raise ValueError("a cell names the budget of labels it learns from") 

59 if not isinstance(seed, int): 

60 raise ValueError("a cell names the seed of its repeat") 

61 return RunCampaignCellCommand( 

62 campaign=CampaignId.parse(campaign), 

63 cell=CampaignCell( 

64 candidate=CandidateRef(candidate), 

65 budget=LabelBudget.parse(budget), 

66 seed=seed, 

67 ),
