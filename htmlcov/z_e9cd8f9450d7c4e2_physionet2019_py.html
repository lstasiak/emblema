39class Physionet2019CorpusReader: 

40 """Reads the ICU stays of the PhysioNet/CinC Challenge 2019, the hourly clinical corpus. 

41 

42 A stay is one ``training_set<X>/p<nnnnnn>.psv`` of pipe-separated rows, one per hour in the 

43 unit: 34 clinical variables as measured in that hour or ``NaN``, six descriptors repeated on 

44 every row, the hour since admission and the challenge's sepsis label. A stay is a unit and a 

45 variable a channel; most cells are empty, so the regime promises no cadence. Time is the hour 

46 since admission, and a stay spans from its first recorded hour to the one past its last. 

47 

48 The descriptors become timeless tokens: age, gender and the hours between hospital and ICU 

49 admission as numbers, the ward as a presence token of one of two unit channels, since the 

50 wards have no order a z-score could respect; a descriptor recorded as ``NaN`` is no feature. 

51 The sepsis label is the downstream task's target: it is checked, never read as a channel. 

52 

53 The challenge published no test set with labels, so the task's frozen side is cut from these 

54 sets by name: stays named in ``excluded`` are out of the units, the counts and the checksum, 

55 and a name matching no stay is refused. The checksum covers the files read, in the order 

56 ``UnitFiles`` visits them. 

57 """ 

58

137 def contents() -> Iterator[bytes]: 

138 for path in self._files.paths(): 

139 content = path.read_bytes() 

140 counts.append(len(self._stay_of(path, content).observations)) 

141 yield content

144 checksum = Checksum.of_chunks(contents())

146 channel_schema=self.SCHEMA, 

147 sampling_regime=SamplingRegime.IRREGULAR,

162 @classmethod 

163 def _stay_of(cls, path: Path, content: bytes) -> _Stay: 

164 """The stay ``content`` holds, checked against the challenge's format. 

165 

166 Raises: 

167 MalformedCorpusDataError: If the header is not the challenge's, a row holds another 

168 number of fields or a value that is not a number, the hours do not increase, a 

169 descriptor changes within the stay, a stay names two wards, or the label is not 

170 a bit. 

171 """ 

172 name = f"{path.parent.name}/{path.name}" 

173 lines = numbered_lines(content.split(b"\n")) 

174 first = next(lines, None) 

175 if first is None: 

176 raise MalformedCorpusDataError(f"{name}: no header") 

177 number, header = first 

178 fields = tuple(fields_of(name, number, header, _SEPARATOR, len(cls.HEADER))) 

179 if fields != cls.HEADER: 

180 raise MalformedCorpusDataError( 

181 f"{name}, line {number}: expected the header {list(cls.HEADER)}, got {list(fields)}" 

182 ) 

183 descriptors: dict[str, str] | None = None 

184 described_at = 0 

185 observations: list[Observation] = [] 

186 first_hour: int | None = None 

187 last_hour: int | None = None 

188 timed = len(cls.SERIES) 

189 for number, line in lines: 

190 row = fields_of(name, number, line, _SEPARATOR, len(cls.HEADER)) 

191 hour = cls._hour_of(name, number, row[-2], last_hour) 

192 first_hour = hour if first_hour is None else first_hour 

193 last_hour = hour 

194 if row[-1] not in ("0", "1"): 

195 raise MalformedCorpusDataError(f"{name}, line {number}: {_LABEL} is not a bit") 

196 given = dict(zip(cls._DESCRIPTORS, row[timed:-2], strict=True)) 

197 if descriptors is None: 

198 descriptors, described_at = given, number 

199 elif given != descriptors: 

200 raise MalformedCorpusDataError( 

201 f"{name}, line {number}: a descriptor changes within the stay" 

202 ) 

203 for channel, value in zip(cls.SERIES, row[:timed], strict=True): 

204 if value != _NOT_RECORDED: 

205 observations.append( 

206 Observation( 

207 channel.name, float(hour), finite_floats(name, number, [value])[0] 

208 ) 

209 ) 

210 if descriptors is None or first_hour is None or last_hour is None: 

211 raise MalformedCorpusDataError(f"{name}: no row") 

212 return _Stay( 

213 cls._features_of(name, described_at, descriptors), 

214 tuple(observations), 

215 TimeExtent(float(first_hour), float(last_hour) + 1.0), 

216 )

218 @staticmethod 

219 def _hour_of(name: str, number: int, cell: str, last: int | None) -> int: 

220 """The hour a row's ``ICULOS`` names: a whole number after the row before it. 

221 

222 Raises: 

223 MalformedCorpusDataError: If the cell is not a whole number or the hour does not 

224 follow the previous row's. 

225 """ 

226 if not (cell.isascii() and cell.isdigit()): 

227 raise MalformedCorpusDataError( 

228 f"{name}, line {number}: {_HOUR} {cell!r} is not a whole number of hours" 

229 ) 

230 hour = int(cell) 

231 if last is not None and hour <= last: 

232 raise MalformedCorpusDataError(f"{name}, line {number}: goes back in time") 

233 return hour

235 @classmethod 

236 def _features_of( 

237 cls, name: str, number: int, descriptors: Mapping[str, str] 

238 ) -> tuple[StaticFeature, ...]: 

239 """The timeless tokens the descriptors first given on line ``number`` become. 

240 

241 Raises: 

242 MalformedCorpusDataError: If a descriptor is not a number, a ward flag is not a bit, 

243 or the stay is flagged as in both wards. 

244 """ 

245 features: list[StaticFeature] = [] 

246 for column, channel in cls.MEASURED_DESCRIPTORS.items(): 

247 value = descriptors[column] 

248 if value != _NOT_RECORDED: 

249 features.append( 

250 StaticFeature(channel.name, finite_floats(name, number, [value])[0]) 

251 ) 

252 wards: list[Channel] = [] 

253 for column, channel in cls.WARDS.items(): 

254 value = descriptors[column] 

255 if value == _NOT_RECORDED: 

256 continue 

257 flag = finite_floats(name, number, [value])[0] 

258 if flag not in (_PRESENT, _ABSENT): 

259 raise MalformedCorpusDataError( 

260 f"{name}, line {number}: {column} {value!r} is not a bit" 

261 ) 

262 if flag == _PRESENT: 

263 wards.append(channel) 

264 if len(wards) > 1: 

265 raise MalformedCorpusDataError(f"{name}, line {number}: a stay in both wards") 

266 features.extend(StaticFeature(ward.name, _PRESENT) for ward in wards) 

267 return tuple(features)
