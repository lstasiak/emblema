5from emblema.pretraining.domain.exceptions import InvalidCorpusShareError 

6from emblema.shared.kernel.ordering import seeded_rank 

7 

8# Frames the ranking apart from the one the Catalog splits units with, so a run whose seed equals 

9# the split's does not read the units that happen to rank just past the validation cut. 

10_RANKING = "training-share" 

11 

12 

13@dataclass(frozen=True, kw_only=True) 

14class CorpusShare: 

15 """How much of a corpus's training side a run reads, and which units that is. 

16 

17 A run over part of a corpus takes whole units, never a share of each unit's windows: windows 

18 of one unit overlap, and a cut through them would put neighbours on both sides of it, as a 

19 split through windows would. The units are ranked by the run's seed, each on its own, and the 

20 share takes the lowest ranks that fit the fraction, rounded up so that a share of one unit is 

21 one unit. Two shares of one seed are therefore nested — a tenth lies inside a quarter, a 

22 quarter inside a half — which is what a curve over fractions wants: each point adds data to the 

23 last rather than drawing afresh. The validation side is never shared: a run over a tenth is 

24 scored on all of it, or its loss would not be comparable with a run over the whole. 

25 

26 Invariants: the fraction lies in ``(0, 1]``. 

27 

28 Attributes: 

29 fraction: Share of the training units read. 

30 seed: Seed the units are ranked by; the run's own. 

31 """ 

32 

33 fraction: float 

34 seed: int 

35 

36 def __post_init__(self) -> None: 

37 if not isfinite(self.fraction) or not 0.0 < self.fraction <= 1.0: 

38 raise InvalidCorpusShareError(f"fraction must lie in (0, 1], got {self.fraction}") 

39 

40 @property 

41 def is_whole(self) -> bool: 

42 return self.fraction == 1.0 

43 

44 def select(self, units: Sequence[str]) -> tuple[str, ...]: 

45 """The units this share reads, in the order they were given. 

46 

47 Raises: 

48 InvalidCorpusShareError: If a unit repeats. 

49 """ 

50 if len(set(units)) != len(units): 

51 raise InvalidCorpusShareError("unit keys must be unique") 

52 if self.is_whole: 

53 return tuple(units) 

54 ranked = sorted(units, key=lambda unit: (seeded_rank(self.seed, _RANKING, unit), unit)) 

55 # Rounded before the ceiling: three tenths of ten units is three, not the four a binary 

56 # 0.3 × 10 would round up to. 

57 chosen = set(ranked[: ceil(round(self.fraction * len(units), 9))]) 

58 return tuple(unit for unit in units if unit in chosen)
