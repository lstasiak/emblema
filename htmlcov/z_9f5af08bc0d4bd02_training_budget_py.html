8@dataclass(frozen=True, kw_only=True) 

9class TrainingBudget: 

10 """How long a run trains, in how large a step, and from which seed. 

11 

12 The batch a gradient is taken over is ``batch_size * accumulation_steps``: accumulation buys 

13 the larger batch on a device that cannot hold it, and is therefore a budget parameter rather 

14 than a property of the model. Epochs, not steps, are the unit here, because the corpus decides 

15 how many batches an epoch holds and the same budget has to describe a run over a tenth of it. 

16 

17 Invariants: every count is positive; the peak rate is positive and finite; the warmup is not 

18 negative and leaves something to decay over, since a run that only warms up never reaches the 

19 floor it states; the floor lies in ``[0, 1]``. 

20 

21 Attributes: 

22 epochs: Passes over the training windows. 

23 batch_size: Windows per micro-batch, which is what a device has to hold. 

24 accumulation_steps: Micro-batches summed into one optimiser step. 

25 learning_rate: Peak rate, which the schedule scales. 

26 warmup_epochs: Epochs the rate climbs to the peak over; a fraction of one where a short 

27 run warms up over part of its first epoch. 

28 final_lr_fraction: Fraction of the peak the decay ends at. 

29 seed: Seed of everything the run draws: weights, order of windows, masks. 

30 """ 

31 

32 epochs: int 

33 batch_size: int 

34 accumulation_steps: int 

35 learning_rate: float 

36 warmup_epochs: float 

37 final_lr_fraction: float 

38 seed: int

62 @property 

63 def effective_batch_size(self) -> int: 

64 """Windows one gradient is taken over.""" 

65 return self.batch_size * self.accumulation_steps 

66 

67 def steps_per_epoch(self, batches: int) -> int: 

68 """Optimiser steps an epoch of ``batches`` micro-batches takes. 

69 

70 A trailing group of fewer micro-batches than the accumulation asks for still steps, so 

71 the last windows of an epoch reach the weights. 

72 

73 Raises: 

74 InvalidTrainingBudgetError: If ``batches`` is not positive. 

75 """ 

76 if batches < 1: 

77 raise InvalidTrainingBudgetError(f"an epoch must hold a batch, got {batches}") 

78 return ceil(batches / self.accumulation_steps) 

79 

80 def takes_a_step_at(self, index: int, batches: int) -> bool: 

81 """Whether the micro-batch at ``index`` of an epoch of ``batches`` applies the gradient. 

82 

83 A group of ``accumulation_steps`` micro-batches makes one step, and the last micro-batch of 

84 an epoch closes whatever group it is in, so the windows at the end of an epoch reach the 

85 weights. Keyed on the index rather than on a counter, so a run resumed inside an epoch 

86 groups its micro-batches exactly as the uninterrupted run did. 

87 

88 Raises: 

89 InvalidTrainingBudgetError: If the index is not one of an epoch of that many batches. 

90 """ 

91 if batches < 1: 

92 raise InvalidTrainingBudgetError(f"an epoch must hold a batch, got {batches}") 

93 if not 0 <= index < batches: 

94 raise InvalidTrainingBudgetError(f"index must lie in [0, {batches}), got {index}") 

95 return (index + 1) % self.accumulation_steps == 0 or index + 1 == batches 

96 

97 def schedule(self, batches: int) -> LearningRateSchedule: 

98 """The rate over the whole run, in optimiser steps, for an epoch of ``batches``. 

99 

100 Raises: 

101 InvalidTrainingBudgetError: If ``batches`` is not positive. 

102 """ 

103 steps = self.steps_per_epoch(batches) 

104 return LearningRateSchedule( 

105 warmup_steps=round(self.warmup_epochs * steps), 

106 total_steps=self.epochs * steps, 

107 final_fraction=self.final_lr_fraction, 

108 )
