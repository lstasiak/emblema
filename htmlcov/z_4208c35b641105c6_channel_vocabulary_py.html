15@dataclass(frozen=True) 

16class VocabularyEntry: 

17 """One channel of one corpus and the identifier tokens carry for it. 

18 

19 Attributes: 

20 channel_id: Identifier the tokens of this channel carry; positive, unique. 

21 corpus: Name of the corpus the channel belongs to, non-blank. 

22 channel: Name of the channel within that corpus, non-blank. 

23 timeless: Whether the channel is a static feature of its units rather than a measurement. 

24 unit: Physical unit of the values, when the source documents one; kept so that a channel can 

25 later be described by more than its identifier. 

26 """ 

27 

28 channel_id: int 

29 corpus: str 

30 channel: str 

31 timeless: bool = False 

32 unit: str | None = None 

33 

34 def __post_init__(self) -> None: 

35 if self.channel_id <= PADDING_CHANNEL_ID: 

36 raise InvalidChannelVocabularyError( 

37 f"channel id must be positive, got {self.channel_id}" 

38 ) 

39 for label, text in (("corpus", self.corpus), ("channel", self.channel)): 

40 if not text or text != text.strip(): 

41 raise InvalidChannelVocabularyError( 

42 f"{label} name must be non-blank without surrounding whitespace" 

43 )

46@dataclass(frozen=True) 

47class ChannelVocabulary: 

48 """The registry that gives every channel of every registered corpus one identifier. 

49 

50 Identifiers are what a model's channel embedding is indexed by, so the vocabulary is a part of 

51 every tokenised corpus and of every model built from one. Two rules make that safe. Spaces are 

52 disjoint per corpus: a channel is a (corpus, name) pair, so ``T2`` of one corpus and ``T2`` of 

53 another never share an identifier. And the registry is append-only: extending it with a corpus 

54 adds entries for the channels it has not seen and changes nothing it has, so identifiers stay 

55 valid for every artifact that already carries them. Identifiers run 1..n in registration 

56 order; 0 is reserved for padding. 

57 

58 Attributes: 

59 entries: Registered channels, entry ``i`` carrying identifier ``i + 1``. 

60 """ 

61 

62 entries: tuple[VocabularyEntry, ...] = () 

63 

64 def __post_init__(self) -> None: 

65 for index, entry in enumerate(self.entries): 

66 if entry.channel_id != index + 1: 

67 raise InvalidChannelVocabularyError( 

68 f"entry {index} must carry id {index + 1}, got {entry.channel_id}" 

69 ) 

70 keys = [(entry.corpus, entry.channel) for entry in self.entries] 

71 if len(set(keys)) != len(keys): 

72 raise InvalidChannelVocabularyError("a channel may be registered once per corpus") 

73 

74 def extended_with(self, corpus: str, schema: ChannelSchema) -> Self: 

75 """The vocabulary after registering the channels of ``schema`` under ``corpus``. 

76 

77 Channels already registered keep their identifiers; new ones are appended in schema order, 

78 which is name order, so two extensions with the same schema produce the same vocabulary. 

79 

80 Raises: 

81 ChannelRedeclaredError: If a channel already registered for the corpus is declared with 

82 another unit or another kind; an identifier cannot change meaning. 

83 """ 

84 known = {(entry.corpus, entry.channel): entry for entry in self.entries} 

85 added: list[VocabularyEntry] = [] 

86 for channel in schema: 

87 existing = known.get((corpus, channel.name)) 

88 if existing is None: 

89 added.append( 

90 VocabularyEntry( 

91 channel_id=len(self.entries) + len(added) + 1, 

92 corpus=corpus, 

93 channel=channel.name, 

94 timeless=channel.timeless, 

95 unit=channel.unit, 

96 ) 

97 ) 

98 elif existing.timeless != channel.timeless or existing.unit != channel.unit: 

99 raise ChannelRedeclaredError( 

100 f"channel {channel.name!r} of corpus {corpus!r} is already registered as " 

101 f"{existing}" 

102 ) 

103 return replace(self, entries=(*self.entries, *added)) 

104 

105 def entry(self, channel_id: int) -> VocabularyEntry: 

106 """The entry behind an identifier. 

107 

108 Raises: 

109 UnknownChannelError: If no entry carries that identifier. 

110 """ 

111 if not 1 <= channel_id <= len(self.entries): 

112 raise UnknownChannelError(f"no channel carries id {channel_id}") 

113 return self.entries[channel_id - 1] 

114 

115 def id_of(self, corpus: str, channel: str) -> int: 

116 """The identifier of a channel of a corpus. 

117 

118 Raises: 

119 UnknownChannelError: If the corpus has no channel of that name registered. 

120 """ 

121 for entry in self.entries: 

122 if entry.corpus == corpus and entry.channel == channel: 

123 return entry.channel_id 

124 raise UnknownChannelError(f"corpus {corpus!r} has no channel {channel!r}") 

125 

126 def entries_of(self, corpus: str) -> tuple[VocabularyEntry, ...]: 

127 return tuple(entry for entry in self.entries if entry.corpus == corpus) 

128 

129 def __len__(self) -> int: 

130 return len(self.entries)
