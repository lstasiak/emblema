9class GriddedTokens: 

10 """Lays a window's readings on equal steps and hands them back as tokens. 

11 

12 What a method on a grid sees, given to a network that reads tokens, so the same network can 

13 be fed the raw readings or the gridded ones and nothing else differs. A reading falls in the 

14 step that holds its instant, by the rule every grid here uses, and where several fall in one 

15 step the latest stands. From a channel's first reading on, every step holds a token at the 

16 step's start carrying the channel's latest reading forward; before it, the channel has no 

17 token, as it has none among the raw readings. Static features are left as they are. 

18 

19 The gap of a gridded token is the time since the channel's latest real reading, in whole 

20 steps: the previous token of a gridded channel is always one step back, so a gap measured to 

21 it would say nothing, while this one is zero exactly where a reading was made and grows while 

22 the value is carried. It is the token's counterpart of the mask beside a grid's values. 

23 """ 

24 

25 def __init__(self, steps: int) -> None: 

26 """Lay windows on ``steps`` equal steps. 

27 

28 Raises: 

29 ValueError: If there is no step. 

30 """ 

31 if steps < 1: 

32 raise ValueError(f"a grid has steps, got {steps}") 

33 self._steps = steps 

34 

35 def of(self, windows: Sequence[TokenWindow]) -> list[TokenWindow]: 

36 """Every window laid on the grid, in the order given.""" 

37 return [self._laid(window) for window in windows] 

38 

39 def _laid(self, window: TokenWindow) -> TokenWindow: 

40 ids = np.asarray(window.channel_ids, dtype=np.int64) 

41 values = np.asarray(window.values, dtype=np.float64) 

42 timeless = np.asarray(window.timeless, dtype=bool) 

43 timed = ~timeless 

44 channels, rows = np.unique(ids[timed], return_inverse=True) 

45 cells = rows * self._steps + steps_of( 

46 np.asarray(window.times, dtype=np.float64)[timed], self._steps 

47 ) 

48 # The latest reading of a step stands: the window is in time order, so the last 

49 # occurrence of every cell is its latest reading. 

50 last = len(cells) - 1 - np.unique(cells[::-1], return_index=True)[1] 

51 read = np.zeros((len(channels), self._steps), dtype=np.float64) 

52 observed = np.zeros((len(channels), self._steps), dtype=bool) 

53 read.flat[cells[last]] = values[timed][last] 

54 observed.flat[cells[last]] = True 

55 latest = np.maximum.accumulate(np.where(observed, np.arange(self._steps), -1), axis=1) 

56 # Step by step, and channel by channel within a step: the canonical order of timed 

57 # tokens, since channels come out of ``unique`` sorted and no two tokens share both. 

58 step, row = np.nonzero(latest.T >= 0) 

59 since = latest[row, step] 

60 # Static features lead a window in canonical order, so they are its first tokens. 

61 statics = int(timeless.sum()) 

62 return TokenWindow( 

63 channel_ids=(*window.channel_ids[:statics], *channels[row].tolist()), 

64 values=(*window.values[:statics], *read[row, since].tolist()), 

65 times=(*window.times[:statics], *(step / self._steps).tolist()), 

66 gaps=(*window.gaps[:statics], *((step - since) / self._steps).tolist()), 

67 timeless=(*window.timeless[:statics], *([False] * len(step))), 

68 )
