7@dataclass(frozen=True, kw_only=True) 

8class MaskingStrategy: 

9 """How much of a window is hidden from the encoder, and in what shapes. 

10 

11 Independent draws per window, and a token is hidden if any reaches it: whole channels, so 

12 a value must be inferred from other channels as with a sensor set never seen; blocks of a 

13 channel's time, so it cannot be read off its neighbours; single tokens, as the minority; the 

14 window's tail across every channel, so the hidden part is the future of what is visible and 

15 cannot be interpolated; and an interval of the window's time across every channel, so what 

16 happened then cannot be read off another channel measured at the same moment. The rates are 

17 the primitives, being what an ablation turns and a vectorised draw needs; the fraction they 

18 add up to is ``expected_ratio``, measured on data rather than promised. 

19 

20 Invariants: every rate lies in ``[0, 1]``; ``block_span`` lies in ``(0, 1]``; a horizon that 

21 is drawn has spans with ``0 < horizon_min_span <= horizon_max_span < 1``, and one that is not 

22 states no spans, since a span nothing draws would be a setting the run does not have; an 

23 interval is held to the same two rules; at least one rate is positive, since a strategy that 

24 hides nothing trains nothing; and no draw hides everything with certainty, since a window 

25 with nothing visible leaves nothing to infer from. 

26 

27 Attributes: 

28 channel_rate: Probability that a channel present in the window is hidden whole. 

29 block_rate: Probability that a channel present in the window loses one block of its time. 

30 block_span: Length of that block as a fraction of the window; the block lies inside the 

31 window. A timeless token has no time and is never inside a block. 

32 token_rate: Probability that a token is hidden on its own. 

33 horizon_rate: Probability that a window loses its tail, every channel at once; zero where 

34 no tail is drawn. 

35 horizon_min_span: Shortest tail, as a fraction of the window. 

36 horizon_max_span: Longest tail; each window's is drawn uniformly between the two, so a 

37 run is not taught one length of future. 

38 interval_rate: Probability that a window loses one interval of its time, every channel at 

39 once; zero where none is drawn. 

40 interval_min_span: Shortest interval, as a fraction of the window. 

41 interval_max_span: Longest interval; each window's is drawn uniformly between the two and 

42 lies inside the window. 

43 """ 

44 

45 channel_rate: float 

46 block_rate: float 

47 block_span: float 

48 token_rate: float 

49 horizon_rate: float = 0.0 

50 horizon_min_span: float = 0.0 

51 horizon_max_span: float = 0.0 

52 interval_rate: float = 0.0 

53 interval_min_span: float = 0.0 

54 interval_max_span: float = 0.0

56 def __post_init__(self) -> None: 

57 for label, rate in ( 

58 ("channel_rate", self.channel_rate), 

59 ("block_rate", self.block_rate), 

60 ("token_rate", self.token_rate), 

61 ("horizon_rate", self.horizon_rate), 

62 ("interval_rate", self.interval_rate), 

63 ): 

64 if not 0.0 <= rate <= 1.0: 

65 raise InvalidMaskingStrategyError(f"{label} must lie in [0, 1], got {rate}") 

66 if not 0.0 < self.block_span <= 1.0: 

67 raise InvalidMaskingStrategyError( 

68 f"block_span must lie in (0, 1], got {self.block_span}" 

69 ) 

70 for label, article, rate, shortest, longest in ( 

71 ("horizon", "a", self.horizon_rate, self.horizon_min_span, self.horizon_max_span), 

72 ("interval", "an", self.interval_rate, self.interval_min_span, self.interval_max_span), 

73 ): 

74 if rate == 0.0: 

75 if shortest != 0.0 or longest != 0.0: 

76 raise InvalidMaskingStrategyError( 

77 f"{article} {label} that is never drawn states no spans, got " 

78 f"[{shortest}, {longest}]" 

79 ) 

80 elif not 0.0 < shortest <= longest < 1.0: 

81 raise InvalidMaskingStrategyError( 

82 f"{label} spans must satisfy 0 < {label}_min_span <= {label}_max_span < 1, " 

83 f"got [{shortest}, {longest}]" 

84 ) 

85 if self.expected_ratio == 0.0: 

86 raise InvalidMaskingStrategyError("a strategy must hide something") 

87 if self.channel_rate == 1.0 or self.token_rate == 1.0 or self.expected_ratio == 1.0: 

88 raise InvalidMaskingStrategyError("a strategy must leave something visible")

100 def draws(self, kind: MaskKind) -> bool: 

101 """Whether the draw a kind of mask is named after can hide anything under this strategy. 

102 

103 A channel can also be left without a visible token by the other draws; that outcome 

104 counts as a hidden channel, but the strategy did not ask for it. 

105 """ 

106 rate = { 

107 MaskKind.CHANNEL: self.channel_rate, 

108 MaskKind.BLOCK: self.block_rate, 

109 MaskKind.TOKEN: self.token_rate, 

110 MaskKind.HORIZON: self.horizon_rate, 

111 MaskKind.INTERVAL: self.interval_rate, 

112 }[kind] 

113 return rate > 0.0

115 @property 

116 def horizon_mean_span(self) -> float: 

117 """The tail a window loses on average, when it loses one.""" 

118 return (self.horizon_min_span + self.horizon_max_span) / 2.0 

119 

120 @property 

121 def interval_mean_span(self) -> float: 

122 """The interval a window loses on average, when it loses one.""" 

123 return (self.interval_min_span + self.interval_max_span) / 2.0 

124 

125 @property 

126 def expected_ratio(self) -> float: 

127 """The fraction of a window's tokens the draws hide together. 

128 

129 Exact for a window whose channels hold equally many tokens spread evenly over its length, 

130 where a token survives every draw with the product of their survival probabilities. Blocks, 

131 the tail and the interval all reach a token by its time, so where more than one is drawn 

132 the product of their averages over time is an approximation. 

133 """ 

134 survives = ( 

135 (1.0 - self.channel_rate) 

136 * (1.0 - self.block_rate * self.block_span) 

137 * (1.0 - self.token_rate) 

138 * (1.0 - self.horizon_rate * self.horizon_mean_span) 

139 * (1.0 - self.interval_rate * self.interval_mean_span) 

140 ) 

141 return 1.0 - survives
