7@dataclass(frozen=True, kw_only=True) 

8class MaskingStrategy: 

9 """How much of a window is hidden from the encoder, and in what shapes. 

10 

11 Independent draws per window, and a token is hidden if any reaches it: whole channels, so 

12 a value must be inferred from other channels as with a sensor set never seen; blocks of a 

13 channel's time, so it cannot be read off its neighbours; single tokens, as the minority; and 

14 the window's tail across every channel, so the hidden part is the future of what is visible 

15 and cannot be interpolated. The rates are the primitives, being what an ablation turns and a 

16 vectorised draw needs; the fraction they add up to is ``expected_ratio``, measured on data 

17 rather than promised. 

18 

19 Invariants: every rate lies in ``[0, 1]``; ``block_span`` lies in ``(0, 1]``; a horizon that 

20 is drawn has spans with ``0 < horizon_min_span <= horizon_max_span < 1``, and one that is not 

21 states no spans, since a span nothing draws would be a setting the run does not have; at 

22 least one rate is positive, since a strategy that hides nothing trains nothing; and no draw 

23 hides everything with certainty, since a window with nothing visible leaves nothing to infer 

24 from. 

25 

26 Attributes: 

27 channel_rate: Probability that a channel present in the window is hidden whole. 

28 block_rate: Probability that a channel present in the window loses one block of its time. 

29 block_span: Length of that block as a fraction of the window; the block lies inside the 

30 window. A timeless token has no time and is never inside a block. 

31 token_rate: Probability that a token is hidden on its own. 

32 horizon_rate: Probability that a window loses its tail, every channel at once; zero where 

33 no tail is drawn. 

34 horizon_min_span: Shortest tail, as a fraction of the window. 

35 horizon_max_span: Longest tail; each window's is drawn uniformly between the two, so a 

36 run is not taught one length of future. 

37 """ 

38 

39 channel_rate: float 

40 block_rate: float 

41 block_span: float 

42 token_rate: float 

43 horizon_rate: float = 0.0 

44 horizon_min_span: float = 0.0 

45 horizon_max_span: float = 0.0 

46 

47 def __post_init__(self) -> None: 

48 for label, rate in ( 

49 ("channel_rate", self.channel_rate), 

50 ("block_rate", self.block_rate), 

51 ("token_rate", self.token_rate), 

52 ("horizon_rate", self.horizon_rate), 

53 ): 

54 if not 0.0 <= rate <= 1.0: 

55 raise InvalidMaskingStrategyError(f"{label} must lie in [0, 1], got {rate}") 

56 if not 0.0 < self.block_span <= 1.0: 

57 raise InvalidMaskingStrategyError( 

58 f"block_span must lie in (0, 1], got {self.block_span}" 

59 ) 

60 if self.horizon_rate == 0.0: 

61 if self.horizon_min_span != 0.0 or self.horizon_max_span != 0.0: 

62 raise InvalidMaskingStrategyError( 

63 "a horizon that is never drawn states no spans, got " 

64 f"[{self.horizon_min_span}, {self.horizon_max_span}]" 

65 ) 

66 elif not 0.0 < self.horizon_min_span <= self.horizon_max_span < 1.0: 

67 raise InvalidMaskingStrategyError( 

68 "horizon spans must satisfy 0 < horizon_min_span <= horizon_max_span < 1, got " 

69 f"[{self.horizon_min_span}, {self.horizon_max_span}]" 

70 ) 

71 if self.expected_ratio == 0.0: 

72 raise InvalidMaskingStrategyError("a strategy must hide something") 

73 if self.channel_rate == 1.0 or self.token_rate == 1.0 or self.expected_ratio == 1.0: 

74 raise InvalidMaskingStrategyError("a strategy must leave something visible") 

75 

76 @property 

77 def has_horizon(self) -> bool: 

78 """Whether any window loses its tail.""" 

79 return self.horizon_rate > 0.0 

80 

81 def draws(self, kind: MaskKind) -> bool: 

82 """Whether the draw a kind of mask is named after can hide anything under this strategy. 

83 

84 A channel can also be left without a visible token by the other draws; that outcome 

85 counts as a hidden channel, but the strategy did not ask for it. 

86 """ 

87 rate = { 

88 MaskKind.CHANNEL: self.channel_rate, 

89 MaskKind.BLOCK: self.block_rate, 

90 MaskKind.TOKEN: self.token_rate, 

91 MaskKind.HORIZON: self.horizon_rate, 

92 }[kind] 

93 return rate > 0.0 

94 

95 @property 

96 def horizon_mean_span(self) -> float: 

97 """The tail a window loses on average, when it loses one.""" 

98 return (self.horizon_min_span + self.horizon_max_span) / 2.0 

99 

100 @property 

101 def expected_ratio(self) -> float: 

102 """The fraction of a window's tokens the draws hide together. 

103 

104 Exact for a window whose channels hold equally many tokens spread evenly over its length, 

105 where a token survives every draw with the product of their survival probabilities. Blocks 

106 and the tail both reach a token by its time, so where both are drawn the product of their 

107 averages over time is an approximation. 

108 """ 

109 survives = ( 

110 (1.0 - self.channel_rate) 

111 * (1.0 - self.block_rate * self.block_span) 

112 * (1.0 - self.token_rate) 

113 * (1.0 - self.horizon_rate * self.horizon_mean_span) 

114 ) 

115 return 1.0 - survives
