10class ContrastiveLoss(nn.Module): 

11 """The normalised-temperature cross-entropy over the two halves of every window of a batch. 

12 

13 Each half of each window is an anchor whose positive is the other half of the same window 

14 and whose distractors are both halves of every other window of the batch (Chen et al. 2020, 

15 NT-Xent). The similarity is the cosine divided by the term's temperature; the loss of an 

16 anchor is the cross-entropy of picking its positive among every candidate but itself. The 

17 windows are of one corpus, as a micro-batch is, so what tells them apart is what varies 

18 within a corpus, not which corpus they come from. 

19 

20 A window whose half holds no token is no pair: it is left out as an anchor and as a 

21 distractor, since the state of nothing is not a window's. One window alone has no 

22 distractor and costs nothing, as ``ContrastiveTerm.chance_of`` states. Summed rather than 

23 averaged, as the reconstruction's scorer sums, so whoever weighs the term divides once. 

24 

25 The arithmetic is done in single precision at least, whatever the projections came in, and 

26 outside any autocast: a softmax over similarities divided by a tenth is where half precision 

27 saturates first. 

28 """ 

29 

30 def __init__(self, term: ContrastiveTerm) -> None: 

31 """Score the pairs at the term's temperature. 

32 

33 Raises: 

34 ValueError: If the term is not stated, since it then has no temperature. 

35 """ 

36 super().__init__() 

37 if not term.is_stated: 

38 raise ValueError("a contrastive loss needs a stated term, which has a temperature") 

39 self.term = term 

40 

41 def summed(self, first: Tensor, second: Tensor, valid: Tensor) -> tuple[Tensor, Tensor]: 

42 """The loss summed over the anchors of the valid windows, and how many anchors there were. 

43 

44 ``first`` and ``second`` are the projections of the two halves, ``[batch, width]`` each; 

45 ``valid`` says which windows are pairs, ``[batch]``. 

46 """ 

47 pairs = int(valid.sum()) 

48 if pairs < 2: 

49 zero = torch.zeros((), dtype=torch.float32, device=first.device) 

50 return zero, torch.zeros((), dtype=torch.int64, device=first.device) 

51 anchors = 2 * pairs 

52 halves = torch.cat([first[valid], second[valid]]).to( 

53 torch.promote_types(first.dtype, torch.float32) 

54 ) 

55 with torch.autocast(device_type=halves.device.type, enabled=False): 

56 normalised = functional.normalize(halves, dim=-1) 

57 similarities = normalised @ normalised.T / self.term.temperature 

58 # An anchor is not its own candidate: its similarity to itself is the largest of 

59 # all and would be the easiest answer. 

60 own = torch.eye(anchors, dtype=torch.bool, device=halves.device) 

61 candidates = similarities.masked_fill(own, -math.inf) 

62 positives = torch.arange(anchors, device=halves.device).roll(pairs) 

63 losses = ( 

64 torch.logsumexp(candidates, dim=1) 

65 - candidates[torch.arange(anchors, device=halves.device), positives] 

66 ) 

67 return losses.sum(), torch.tensor(anchors, device=first.device) 

68 

69 def chance(self, valid: Tensor) -> Tensor: 

70 """What the same anchors cost a model whose similarities are all equal, summed.""" 

71 pairs = int(valid.sum()) 

72 anchors = 2 * pairs if pairs >= 2 else 0 

73 return torch.tensor( 

74 anchors * ContrastiveTerm.chance_of(pairs), dtype=torch.float32, device=valid.device 

75 )
