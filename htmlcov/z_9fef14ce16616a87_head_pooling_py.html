10class PoolingScheme(StrEnum): 

11 """How the states of a window's tokens become the one state a head reads. 

12 

13 Every scheme here is free of the channel layout: it weighs tokens by where they sit in the 

14 window or by what they hold, never by which channel they came from, so a head under any of 

15 them reads a corpus of eight channels as it reads one of forty-four. A pooling laid out per 

16 channel would be as wide as the corpus and is left to the classical baselines. 

17 

18 Attributes: 

19 MEAN: The mean over every observed token of the window. 

20 TAIL: The mean over the observed tokens in the last share of the window, the window's 

21 static features included, since a static feature is current at every instant. 

22 ATTENTION: A weighted mean whose weights a learnt query reads off the states themselves, 

23 so the head can learn which part of the window to read from. 

24 """ 

25 

26 MEAN = "mean" 

27 TAIL = "tail" 

28 ATTENTION = "attention" 

29 

30 @property 

31 def learns_weights(self) -> bool: 

32 """Whether the pooling has weights of its own that a run trains.""" 

33 return self is PoolingScheme.ATTENTION

36@dataclass(frozen=True, kw_only=True) 

37class HeadPooling: 

38 """The pooling of a network's head: which scheme, and how much of the window a tail keeps. 

39 

40 Stated once for every network a campaign compares, so a set encoder and a patch model pool 

41 by one vocabulary and a variant name turns the same knobs on both. A share under a scheme 

42 that reads none would be a knob recorded as turned and changing nothing, so it is refused 

43 rather than normalised. A tail over the whole window computes the mean and is allowed: a 

44 variant turns one knob at a time, and the tail has to exist before its share is set. 

45 

46 Invariants: the share lies in ``(0, 1]``, and is one unless the scheme is a tail. 

47 

48 Attributes: 

49 pooling: Which scheme. 

50 tail_share: Share of the window's length a tail keeps, from its end; one otherwise. 

51 """ 

52 

53 pooling: PoolingScheme 

54 tail_share: float = 1.0 

55 

56 KNOBS: ClassVar[tuple[str, ...]] = ("pooling", "tail_share") 

57 

58 def __post_init__(self) -> None: 

59 if not isfinite(self.tail_share) or not 0.0 < self.tail_share <= 1.0: 

60 raise InvalidHeadPoolingError(f"tail_share must lie in (0, 1], got {self.tail_share}") 

61 if self.tail_share < 1.0 and self.pooling is not PoolingScheme.TAIL: 

62 raise InvalidHeadPoolingError( 

63 f"{self.pooling} reads the whole window, so a share of it is no knob" 

64 )

67 def mean(cls) -> Self:

69 return cls(pooling=PoolingScheme.MEAN)

72 """This pooling with ``knob`` turned to ``value``. 

73 

74 Raises:

76 """ 

77 if knob not in self.KNOBS: 

78 raise UnknownKnobError(f"a pooling has no knob {knob!r}; it turns {self.KNOBS}") 

79 return turned(self, knob, value)
