7@dataclass(frozen=True) 

8class ChannelStatistics: 

9 """What the training data established about one channel, and the normalisation it implies. 

10 

11 The statistics are facts about the data a scheme was fitted on — how many values, their mean 

12 and their population standard deviation — and are kept as such rather than folded into a 

13 transform, because they are also the provenance of the model and the only description of a 

14 channel the data itself offers. Normalisation is the z-score. A channel that never varied in 

15 training keeps its deviations in raw units instead of dividing by a vanishing spread: a value 

16 that does deviate later then stays finite and honest about how far it strayed. 

17 

18 Attributes: 

19 count: Values the statistics were computed over; at least one. 

20 mean: Arithmetic mean; finite. 

21 std: Population standard deviation; finite, non-negative. 

22 """ 

23 

24 count: int 

25 mean: float 

26 std: float 

27 

28 def __post_init__(self) -> None: 

29 if self.count < 1: 

30 raise InvalidChannelStatisticsError(f"count must be positive, got {self.count}") 

31 if not math.isfinite(self.mean): 

32 raise InvalidChannelStatisticsError(f"mean must be finite, got {self.mean}") 

33 if not (math.isfinite(self.std) and self.std >= 0): 

34 raise InvalidChannelStatisticsError( 

35 f"std must be finite and non-negative, got {self.std}" 

36 ) 

37 

38 def normalise(self, value: float) -> float: 

39 return (value - self.mean) / self._scale 

40 

41 def denormalise(self, value: float) -> float: 

42 """The raw value a normalised one was made from. 

43 

44 The inverse of ``normalise`` in arithmetic, not in bits: a value that goes out and comes 

45 back differs in its last places, so whoever compares the two compares them to a tolerance. 

46 """ 

47 return value * self._scale + self.mean 

48 

49 @property 

50 def _scale(self) -> float: 

51 """What a deviation is measured in; one raw unit for a channel that never varied.""" 

52 return self.std if self.std > 0.0 else 1.0
