24_ENCODING = "ascii"

27class CmapssCorpusReader: 

28 """Reads the training trajectories of NASA's C-MAPSS turbofan degradation data set. 

29 

30 Each ``train_FD00x.txt`` holds a row per engine cycle: unit, cycle, three operational settings 

31 and 21 sensors. An engine is a unit and a sensor a channel; cycles are evenly spaced, so the 

32 regime is regular, and the operational settings are not channels. Only training trajectories are 

33 read: the official test engines and their RUL targets are evaluation data outside the Catalog, 

34 so a version's checksum also proves that no test engine was pretrained on. 

35 

36 The reader is bound to a selection of the four subsets, each one operating regime and a corpus 

37 in its own right. Units are keyed ``<subset>/<engine>``, since engine numbers restart per 

38 subset, and an engine of ``L`` cycles spans ``[1, L + 1)``. An engine's rows must form one 

39 contiguous block of its file. The checksum covers the selected files' bytes in subset order, 

40 whatever order the subsets were named in. 

41 """ 

42 

43 SUBSETS: ClassVar[tuple[str, ...]] = ("FD001", "FD002", "FD003", "FD004") 

44 # The 21 sensor outputs in file column order, named and united as in Saxena, Goebel, Simon and 

45 # Eklund (2008), "Damage Propagation Modeling for Aircraft Engine Run-to-Failure Simulation". 

46 # A ratio has no unit. 

47 SENSORS: ClassVar[tuple[Channel, ...]] = ( 

48 Channel("T2", "°R"), 

49 Channel("T24", "°R"), 

50 Channel("T30", "°R"), 

51 Channel("T50", "°R"), 

52 Channel("P2", "psia"), 

53 Channel("P15", "psia"), 

54 Channel("P30", "psia"), 

55 Channel("Nf", "rpm"), 

56 Channel("Nc", "rpm"), 

57 Channel("epr"), 

58 Channel("Ps30", "psia"), 

59 Channel("phi", "pps/psi"), 

60 Channel("NRf", "rpm"), 

61 Channel("NRc", "rpm"), 

62 Channel("BPR"), 

63 Channel("farB"), 

64 Channel("htBleed"), 

65 Channel("Nf_dmd", "rpm"), 

66 Channel("PCNfR_dmd", "rpm"), 

67 Channel("W31", "lbm/s"), 

68 Channel("W32", "lbm/s"), 

69 ) 

70 _WIDTH: ClassVar[int] = _LEADING_COLUMNS + len(SENSORS)

72 def __init__(self, root: Path, subsets: Iterable[str] = SUBSETS) -> None: 

73 chosen = chosen_subsets(subsets, self.SUBSETS, "C-MAPSS subset") 

74 self._files = {name: root / f"train_{name}.txt" for name in chosen} 

75 

76 def describe(self) -> CorpusDescription: 

77 lengths: list[int] = [] 

78 

79 def contents() -> Iterator[bytes]: 

80 for path in self._files.values(): 

81 content = self._bytes_of(path) 

82 lengths.extend(self._cycles_per_engine(path.name, content).values()) 

83 yield content 

84 

85 # The checksum consumes the files one at a time, so the whole corpus is never in memory. 

86 checksum = Checksum.of_chunks(contents()) 

87 return CorpusDescription( 

88 channel_schema=ChannelSchema(frozenset(self.SENSORS)), 

89 sampling_regime=SamplingRegime.REGULAR, 

90 content=CorpusContent( 

91 checksum=checksum, 

92 unit_count=len(lengths), 

93 observation_count=sum(lengths) * len(self.SENSORS), 

94 ), 

95 )

97 def read_units(self) -> Iterator[CorpusUnit]: 

98 for name, path in self._files.items(): 

99 engines = self._cycles_per_engine(path.name, self._bytes_of(path)) 

100 for engine, cycles in engines.items(): 

101 yield CorpusUnit(UnitKey(f"{name}/{engine}"), TimeExtent(1.0, cycles + 1.0))

103 def read_observations(self, unit: UnitKey) -> Iterator[Observation]: 

104 """One engine's sensor values, cycle by cycle. 

105 

106 A key that could name no engine is refused before the stream starts; a file that turns 

107 out not to hold the engine is reported when the stream ends, because only a scan can tell. 

108 """ 

109 name, engine = self._locate(unit) 

110 path = self._files[name] 

111 if not path.is_file(): 

112 raise CorpusDataNotFoundError(f"{path} is missing") 

113 return self._observations_of(path, engine) 

114 

115 @classmethod 

116 def _observations_of(cls, path: Path, engine: int) -> Iterator[Observation]: 

117 marker = str(engine) 

118 expected = 1 

119 with path.open("rb") as file: 

120 for number, line in numbered_lines(file, _ENCODING): 

121 if line.split(None, 1)[0] != marker: 

122 if expected > 1: 

123 break 

124 continue 

125 _, cycle, sensors = cls._parse_row(path.name, number, line) 

126 if cycle != expected: 

127 raise MalformedCorpusDataError( 

128 f"{path.name}, line {number}: engine {engine} jumps to cycle {cycle} after " 

129 f"cycle {expected - 1}" 

130 ) 

131 expected += 1 

132 for channel, value in zip(cls.SENSORS, sensors, strict=True): 

133 yield Observation(channel.name, float(cycle), value) 

134 if expected == 1: 

135 raise UnknownUnitError(f"{path.name} has no engine {engine}")

137 def _locate(self, unit: UnitKey) -> tuple[str, int]: 

138 name, separator, engine = unit.value.partition("/") 

139 if not separator or name not in self._files or not (engine.isascii() and engine.isdigit()): 

140 raise UnknownUnitError(f"{unit} is not an engine of a selected subset") 

141 return name, int(engine) 

142 

143 @staticmethod 

144 def _bytes_of(path: Path) -> bytes: 

145 if not path.is_file(): 

146 raise CorpusDataNotFoundError(f"{path} is missing") 

147 return path.read_bytes() 

148 

149 @classmethod 

150 def _cycles_per_engine(cls, name: str, content: bytes) -> dict[int, int]: 

151 cycles: dict[int, int] = {} 

152 current: int | None = None 

153 for number, line in numbered_lines(content.split(b"\n"), _ENCODING): 

154 engine, cycle, _ = cls._parse_row(name, number, line) 

155 if engine != current: 

156 if engine in cycles: 

157 raise MalformedCorpusDataError( 

158 f"{name}, line {number}: engine {engine} resumes after engine {current}" 

159 ) 

160 current = engine 

161 expected = cycles.get(engine, 0) + 1 

162 if cycle != expected: 

163 raise MalformedCorpusDataError( 

164 f"{name}, line {number}: engine {engine} jumps to cycle {cycle} after " 

165 f"cycle {expected - 1}" 

166 ) 

167 cycles[engine] = cycle 

168 if not cycles: 

169 raise MalformedCorpusDataError(f"{name}: no engine") 

170 return cycles

172 @classmethod 

173 def _parse_row(cls, name: str, number: int, line: str) -> tuple[int, int, list[float]]: 

174 fields = fields_of(name, number, line, None, cls._WIDTH) 

175 try: 

176 engine, cycle = int(fields[0]), int(fields[1]) 

177 except ValueError as error: 

178 raise MalformedCorpusDataError(f"{name}, line {number}: {error}") from error 

179 # The operational settings are held to being numbers like the sensors, and then left out. 

180 values = finite_floats(name, number, fields[2:]) 

181 return engine, cycle, values[_LEADING_COLUMNS - 2 :]
