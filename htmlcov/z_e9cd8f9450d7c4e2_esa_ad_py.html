37class EsaAdCorpusReader: 

38 """Reads the training halves of the ESA Anomaly Dataset, the satellite telemetry benchmark. 

39 

40 A mission is a directory of ``channels/channel_<n>.zip``, one pickled pandas data frame per 

41 channel: instants as the index, the values as its one column. Only the benchmark's lightweight 

42 channels are read, only the first half of a mission by time — the benchmark's training 

43 portion, cut back to a thirty-second bin boundary — and within it the earliest observation of 

44 each channel in each bin, at its native instant, so the volume is bounded without a grid. 

45 

46 Units are the calendar months of a mission's training half, keyed ``<mission>/<YYYY-MM>``: the 

47 two missions are too few to split on and have channels of their own (ADR-0025). A month is a 

48 unit whether or not any telemetry reached the ground in it. Channels are named 

49 ``<mission>/channel_<n>``, since a channel number names a different quantity on each 

50 satellite. Observations are placed in hours since the mission's start, the whole second of 

51 its first kept observation, on one axis shared by the mission's months; the regime promises 

52 no cadence. 

53 

54 The checksum covers the selected missions' channel archives in mission order and 

55 channel-number order, whatever order the missions were named in. Unpickling runs the stream it 

56 reads, so the files are trusted as far as the fetch script's checksums vouch for them. The 

57 mission last read is kept, so that its months are served from memory rather than by 

58 unpickling a gigabyte each. 

59 """ 

60 

61 SUBSETS: ClassVar[tuple[str, ...]] = ("ESA-Mission1", "ESA-Mission2") 

62 # The channels the benchmark's lightweight configuration monitors, one run of channel numbers 

63 # per mission: subsystem 5 of Mission1 and subsystem 1 of Mission2. The third published mission 

64 # is left out as the benchmark leaves it out; adding it is an entry here and a decision on its 

65 # channels. 

66 LIGHTWEIGHT: ClassVar[Mapping[str, range]] = { 

67 "ESA-Mission1": range(41, 47), 

68 "ESA-Mission2": range(18, 29), 

69 }

71 def __init__(self, root: Path, subsets: Iterable[str] = SUBSETS) -> None: 

72 self._root = root 

73 self._subsets = chosen_subsets(subsets, self.SUBSETS, "ESA mission") 

74 self._loaded: _Mission | None = None

76 def describe(self) -> CorpusDescription: 

77 units = 0 

78 observations = 0 

79 

80 def contents() -> Iterator[bytes]: 

81 nonlocal units, observations 

82 for name in self._subsets: 

83 series = [] 

84 for number, path in self._channel_paths(name): 

85 yield from self._chunks_of(path) 

86 series.append((number, self._series_of(path))) 

87 mission = _Mission.assemble(name, series) 

88 units += len(mission.segments) 

89 observations += mission.observation_count 

90 

91 # The checksum consumes the files a chunk at a time, so no archive is ever held whole. 

92 checksum = Checksum.of_chunks(contents()) 

93 return CorpusDescription( 

94 channel_schema=ChannelSchema( 

95 frozenset( 

96 Channel(_channel_name(name, number)) 

97 for name in self._subsets 

98 for number in self.LIGHTWEIGHT[name] 

99 ) 

100 ), 

101 sampling_regime=SamplingRegime.IRREGULAR, 

102 content=CorpusContent( 

103 checksum=checksum, unit_count=units, observation_count=observations 

104 ), 

105 )

116 def read_observations(self, unit: UnitKey) -> Iterator[Observation]: 

117 name, separator, _ = unit.value.partition("/") 

118 if not separator or name not in self._subsets: 

119 raise UnknownUnitError(f"{unit} is not a month of a selected mission") 

120 mission = self._mission(name) 

121 segment = mission.segment_keyed(unit) 

122 if segment is None:

135 def _channel_paths(self, mission: str) -> list[tuple[int, Path]]: 

136 """The lightweight channel archives of a mission in channel-number order.""" 

137 folder = self._root / mission / _CHANNELS_FOLDER 

138 if not folder.is_dir(): 

139 raise CorpusDataNotFoundError(f"{folder} is missing") 

140 paths = [(number, folder / f"channel_{number}.zip") for number in self.LIGHTWEIGHT[mission]] 

141 for _, path in paths: 

142 if not path.is_file(): 

143 raise CorpusDataNotFoundError(f"{path} is missing") 

144 return paths

146 @staticmethod 

147 def _chunks_of(path: Path) -> Iterator[bytes]: 

148 with path.open("rb") as archive: 

149 while chunk := archive.read(_CHUNK_BYTES): 

150 yield chunk

152 @staticmethod 

153 def _series_of(path: Path) -> "_Series": 

154 """The instants and values of one channel archive, validated against its format.""" 

155 # Imported here: the extra that provides pandas is needed by this reader alone, and a 

156 # process publishing another corpus must not need it. 

157 import pandas as pd 

158 

159 name = path.name 

160 try: 

161 with zipfile.ZipFile(path) as archive: 

162 members = archive.namelist() 

163 if len(members) != 1: 

164 raise MalformedCorpusDataError( 

165 f"{name}: expected an archive of one member, got {members}" 

166 ) 

167 with archive.open(members[0]) as member: 

168 frame = pd.read_pickle(member) 

169 except zipfile.BadZipFile as error: 

170 raise MalformedCorpusDataError(f"{name}: not a zip archive") from error 

171 except MalformedCorpusDataError: 

172 raise 

173 # A pickle is a program that builds its object, so a stream that is not the expected one 

174 # fails wherever it was, with whatever error the half-built object raised. 

175 except Exception as error: 

176 raise MalformedCorpusDataError(f"{name}: not a pickled data frame: {error}") from error 

177 if not isinstance(frame, pd.DataFrame) or frame.shape[1] != 1: 

178 raise MalformedCorpusDataError( 

179 f"{name}: expected a data frame of one column, got {type(frame).__name__}" 

180 + (f" of {frame.shape[1]} columns" if isinstance(frame, pd.DataFrame) else "") 

181 ) 

182 if not isinstance(frame.index, pd.DatetimeIndex): 

183 raise MalformedCorpusDataError( 

184 f"{name}: expected instants as the index, got {type(frame.index).__name__}" 

185 ) 

186 if frame.index.hasnans: 

187 raise MalformedCorpusDataError(f"{name}: an instant is missing") 

188 times = frame.index.values.astype("datetime64[ns]").astype(np.int64) 

189 if np.any(times[1:] < times[:-1]): 

190 raise MalformedCorpusDataError(f"{name}: goes back in time") 

191 values = frame.iloc[:, 0].to_numpy() 

192 if not np.issubdtype(values.dtype, np.number): 

193 raise MalformedCorpusDataError(f"{name}: values are {values.dtype}, not numbers") 

194 if not np.all(np.isfinite(values)): 

195 raise MalformedCorpusDataError(f"{name}: non-finite value") 

196 return _Series(times, values)

210 def kept_before(self, end: int) -> "_Series": 

211 """The earliest observation of every bin that starts before ``end``.""" 

212 count = int(np.searchsorted(self.times, end, side="left")) 

213 times = self.times[:count] 

214 bins = times // _BIN_NANOSECONDS 

215 first = np.ones(count, dtype=bool) 

216 first[1:] = bins[1:] != bins[:-1] 

217 return _Series(times[first], self.values[:count][first])

229@dataclass(frozen=True) 

230class _Mission: 

231 """A mission's training half, decimated: the arrays every month of it is served from. 

232 

233 ``start`` is the origin of the mission's axis, the whole second of its first observation, and 

234 ``end`` the bin boundary at or before the exact half of the span the selected channels cover, 

235 both in nanoseconds since the epoch. 

236 """ 

237 

238 name: str 

239 start: int 

240 end: int 

241 channels: tuple[tuple[str, _Series], ...] 

242 segments: tuple[_Segment, ...] 

243 

244 @classmethod 

245 def assemble(cls, name: str, series: Iterable[tuple[int, _Series]]) -> "_Mission": 

246 """Cut the channels to the training half and lay the months over it.""" 

247 series = list(series) 

248 spanned = [channel for _, channel in series if len(channel.times)] 

249 if not spanned: 

250 raise MalformedCorpusDataError(f"{name}: no observation in any selected channel") 

251 first = min(int(channel.times[0]) for channel in spanned) // _NANOSECONDS_PER_SECOND 

252 last = max(int(channel.times[-1]) for channel in spanned) // _NANOSECONDS_PER_SECOND 

253 cutoff = first + (last - first) // 2 

254 end_second = cutoff // _BIN_SECONDS * _BIN_SECONDS 

255 if end_second <= first: 

256 raise MalformedCorpusDataError( 

257 f"{name}: spans {last - first + 1} seconds, too few to hold a training half" 

258 ) 

259 start = first * _NANOSECONDS_PER_SECOND 

260 end = end_second * _NANOSECONDS_PER_SECOND 

261 return cls( 

262 name=name, 

263 start=start, 

264 end=end, 

265 channels=tuple( 

266 (_channel_name(name, number), channel.kept_before(end)) 

267 for number, channel in series 

268 ), 

269 segments=cls._months(name, start, end), 

270 ) 

271 

272 @staticmethod 

273 def _months(name: str, start: int, end: int) -> tuple[_Segment, ...]: 

274 first = np.datetime64(start, "ns").astype("datetime64[M]") 

275 last = np.datetime64(end - 1, "ns").astype("datetime64[M]") 

276 beyond = last + np.timedelta64(1, "M") 

277 months = np.arange(first, beyond) 

278 edges = np.arange(first, beyond + np.timedelta64(1, "M")).astype("datetime64[ns]") 

279 return tuple( 

280 _Segment(UnitKey(f"{name}/{month}"), max(int(lower), start), min(int(upper), end)) 

281 for month, lower, upper in zip( 

282 months, edges[:-1].astype(np.int64), edges[1:].astype(np.int64), strict=True 

283 ) 

284 )

286 @property 

287 def observation_count(self) -> int: 

288 return sum(len(channel.times) for _, channel in self.channels)

290 def hours(self, instant: int) -> float: 

291 """An instant on the mission's axis, by the arithmetic that places the observations too. 

292 

293 One division for a month's boundary and for an observation on it, so that the two come 

294 out as the same number and the observation lies inside the month by construction. 

295 """ 

296 return float(np.int64(instant - self.start) / _NANOSECONDS_PER_HOUR)

298 def segment_keyed(self, key: UnitKey) -> _Segment | None: 

299 return next((segment for segment in self.segments if segment.key == key), None)

301 def observations_of(self, segment: _Segment) -> Iterator[Observation]: 

302 """The month's observations of every channel, merged by instant, channels in order.""" 

303 times = [] 

304 values = [] 

305 sources = [] 

306 for index, (_, channel) in enumerate(self.channels): 

307 lower, upper = np.searchsorted(channel.times, [segment.start, segment.end], side="left") 

308 times.append(channel.times[lower:upper]) 

309 values.append(channel.values[lower:upper].astype(np.float64)) 

310 sources.append(np.full(upper - lower, index, dtype=np.intp)) 

311 instants = np.concatenate(times) 

312 order = np.argsort(instants, kind="stable") 

313 hours = ((instants[order] - self.start) / _NANOSECONDS_PER_HOUR).tolist() 

314 names = [self.channels[index][0] for index in np.concatenate(sources)[order].tolist()] 

315 for name, time, value in zip( 

316 names, hours, np.concatenate(values)[order].tolist(), strict=True 

317 ): 

318 yield Observation(name, time, value)
