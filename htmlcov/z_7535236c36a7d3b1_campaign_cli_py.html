23class CampaignCli: 

24 """Command line of a comparison: define the task, declare the campaign, hand out its cells. 

25 

26 Four invocations, and none of them runs a cell. On the machine that keeps the registry:: 

27 

28 uv run python -m emblema.entrypoints.cli.campaign define-task 

29 --corpus <manifest key> <checksum> --task turbofan-fd001 

30 

31 which prints the task's identifier. Then, over a file committed before anything runs:: 

32 

33 uv run python -m emblema.entrypoints.cli.campaign define 

34 --file campaigns/baselines-fd001.toml --task <task id> 

35 

36 which prints the campaign's identifier, and:: 

37 

38 uv run python -m emblema.entrypoints.cli.campaign advance --campaign <campaign id> 

39 

40 which submits every cell that has not run and prints how many it handed over. Running them 

41 is the workers' business: each cell goes to the queue its candidate belongs to, and a grid 

42 resumed after a crash costs a second `advance` and nothing else. Once it has closed:: 

43 

44 uv run python -m emblema.entrypoints.cli.campaign announce --campaign <campaign id> 

45 

46 publishes its conclusion again, for when the delivery made on closing failed, and prints 

47 the checksum of every artifact it kept — what a promotion names. 

48 

49 The design comes from the file and not from flags, which is the whole reason there is a 

50 file. Budgets, seeds and the rule for what counts as a difference passed on a command line 

51 are a decision nobody can date; written down and committed first, they are a registration. 

52 

53 Standard output is the identifiers a person copies into the next invocation, and nothing 

54 else: whatever a library prints while a use case runs goes to standard error. 

55 """

84 def execute( 

85 self, invocation: CampaignInvocation, adapters: Adapters, services: Services 

86 ) -> str: 

87 """Carry out the invocation and return what the process prints for it. 

88 

89 Raises: 

90 SystemExit: If the invocation names a task or a campaign nothing knows, asks for 

91 something this command line does not do, or is refused by the context, with the 

92 reason as the message: a refusal is an answer to the person at the keyboard, not 

93 a crash to be read in a traceback. 

94 """ 

95 try: 

96 match invocation.what: 

97 case self.DEFINE_TASK: 

98 task, corpus = self._known(invocation.task), self._named(invocation.corpus) 

99 sides = adapters.corpus.describe(corpus) 

100 return str(services.define_downstream_task(task.defined_over(corpus, sides))) 

101 case self.DEFINE: 

102 # The task is settled before the file is opened: an invocation that names 

103 # none is refused without anything on disk being read. 

104 over = self._task(invocation.task) 

105 declared = CampaignFile.load(self._named(invocation.file)) 

106 return str(services.define_campaign(self._design(declared, over))) 

107 case self.ADVANCE: 

108 submitted = services.advance_campaign( 

109 AdvanceCampaignCommand(campaign=self._campaign(invocation.campaign)) 

110 ) 

111 return str(submitted) 

112 case self.ANNOUNCE: 

113 announced = services.announce_campaign( 

114 AnnounceCampaignCommand(campaign=self._campaign(invocation.campaign)) 

115 ) 

116 return "\n".join( 

117 str(candidate.artifact.checksum) 

118 for candidate in announced.candidates 

119 if candidate.artifact is not None 

120 ) 

121 case _: 

122 raise SystemExit(f"this command line does not {invocation.what!r}") 

123 except EvaluationError as refusal: 

124 raise SystemExit(str(refusal)) from refusal
