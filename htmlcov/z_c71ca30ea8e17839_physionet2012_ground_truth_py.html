14class Physionet2012GroundTruth: 

15 """What each intensive-care stay recorded for one outcome, read from the challenge's files. 

16 

17 The challenge publishes the outcomes of every stay of a set in one file beside it — 

18 ``Outcomes-a.txt`` for ``set-a`` — a row per stay keyed by its record number, one column per 

19 outcome. Which column is the truth is the scheme's outcome, named once with the task, so the 

20 task and the answers it is given cannot drift apart. Every window of a stay is answered with 

21 the stay's outcome: the truth of this task is a fact about the stay, not about a moment in it. 

22 

23 Units are named ``set-<x>/<RecordID>``, as the reader of this corpus names them. A set's file 

24 is read once and kept. 

25 """ 

26 

27 _RECORD: ClassVar[str] = "RecordID" 

28 _SET_PREFIX: ClassVar[str] = "set-" 

29 

30 def __init__(self, root: Path, scheme: OutcomeScheme) -> None: 

31 self._root = root 

32 self._outcome = scheme.outcome 

33 self._sets: dict[str, dict[str, float]] = {} 

34 

35 def truths_of(self, corpus: str, windows: Sequence[TaskWindow]) -> Mapping[TaskWindow, float]: 

36 # This adapter is the truth of one corpus, so the name is the composite's business. 

37 recorded = {unit: self._recorded(unit) for unit in {window.unit for window in windows}} 

38 return {window: recorded[window.unit] for window in windows} 

39 

40 def _recorded(self, unit: UnitKey) -> float: 

41 subset, _, record = str(unit).partition("/") 

42 if not subset.startswith(self._SET_PREFIX) or not record: 

43 raise UnknownGroundTruthError(f"{unit} is not named set-<x>/<RecordID>") 

44 outcomes = self._of_set(subset) 

45 if record not in outcomes: 

46 raise UnknownGroundTruthError(f"{unit} has no outcome recorded in its set") 

47 return outcomes[record] 

48 

49 def _of_set(self, subset: str) -> dict[str, float]: 

50 if subset not in self._sets: 

51 letter = subset.removeprefix(self._SET_PREFIX) 

52 self._sets[subset] = self._read(self._root / f"Outcomes-{letter}.txt") 

53 return self._sets[subset] 

54 

55 def _read(self, path: Path) -> dict[str, float]: 

56 try: 

57 lines = path.read_text(encoding="utf-8").splitlines() 

58 except OSError as error: 

59 raise UnreadableGroundTruthError(f"cannot read {path}: {error}") from error 

60 if not lines: 

61 raise UnreadableGroundTruthError(f"{path} is empty") 

62 header = lines[0].split(",") 

63 for needed in (self._RECORD, self._outcome): 

64 if needed not in header: 

65 raise UnreadableGroundTruthError(f"{path} has no column {needed}") 

66 record_at, outcome_at = header.index(self._RECORD), header.index(self._outcome) 

67 outcomes: dict[str, float] = {} 

68 for number, line in enumerate(lines[1:], start=2): 

69 if not line.strip(): 69 ↛ 70line 69 didn't jump to line 70 because the condition on line 69 was never true

70 continue 

71 fields = line.split(",") 

72 if len(fields) != len(header): 

73 raise UnreadableGroundTruthError( 

74 f"{path} line {number} has {len(fields)} fields, not {len(header)}" 

75 ) 

76 record = fields[record_at] 

77 if not record.isdigit() or record in outcomes: 

78 raise UnreadableGroundTruthError( 

79 f"{path} line {number} names record {record!r}, which is not a new number" 

80 ) 

81 try: 

82 outcomes[record] = float(fields[outcome_at]) 

83 except ValueError as error: 

84 raise UnreadableGroundTruthError( 

85 f"{path} line {number} records {fields[outcome_at]!r} for {self._outcome}" 

86 ) from error 

87 return outcomes
