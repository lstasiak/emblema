46class TorchDetectionRuntime: 

47 """Fits a detector over the states of a frozen encoder, in this process, on a device. 

48 

49 Every window of the fitting side and of the scored side is laid out as the encoder was 

50 trained to read it, embedded once under no gradient, and the detector is fitted and scored 

51 over the states on the host in double precision: the arithmetic of a distance to the nearest 

52 neighbours is numpy's, as it is for the summaries, so the two detectors differ in what they 

53 read and in nothing else. The seconds an outcome reports start once the block is at hand. 

54 """

140 def _states( 

141 self, embedding: FrozenEmbedding, layout: WindowLayout, windows: Sequence[TokenWindow] 

142 ) -> NDArray[np.float64]: 

143 """One state per window, on the host in double precision, in the order given.""" 

144 laid: list[LaidWindow] = layout.lay(windows) 

145 states = [] 

146 for start in range(0, len(laid), _BATCH): 

147 batch = TokenTensors.of(layout.batch(laid[start : start + _BATCH])).to(self._device) 

148 # Moved before it is widened: an accelerator without double precision hands back 

149 # zeros when asked to do both at once. 

150 states.append(embedding.embed(batch).to("cpu").to(torch.float64).numpy()) 

151 return np.concatenate(states)
