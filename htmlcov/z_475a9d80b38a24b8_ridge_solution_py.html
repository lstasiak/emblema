12@dataclass(frozen=True) 

13class RidgeSolution: 

14 """A linear head solved in closed form over pooled states, its penalty chosen leave-one-out. 

15 

16 Ridge with an unpenalised intercept, the columns scaled by their spread so that one penalty 

17 means the same for every state, solved once through a singular value decomposition and 

18 read off for every penalty in turn: the leave-one-out residuals of a linear smoother are 

19 the residuals divided by one minus the leverage, so the choice costs no refit. The penalty 

20 with the smallest mean squared leave-one-out error wins, the smaller one on a tie. Solved 

21 on the host in double precision, since the states come off an accelerator that may have 

22 none, and moved there before they are widened. The head of a quantity; an outcome's is 

23 ``LogisticSolution``. 

24 

25 Attributes: 

26 weights: The linear map over the states as they were given, one weight per column. 

27 intercept: What the map adds. 

28 penalty: The penalty the fit chose. 

29 leave_one_out: Mean squared leave-one-out error under each penalty, in their order. 

30 """ 

31 

32 weights: Tensor 

33 intercept: float 

34 penalty: float 

35 leave_one_out: tuple[float, ...] 

36 

37 @classmethod 

38 def fitted(cls, states: Tensor, targets: Tensor, penalties: RidgePenalties) -> Self: 

39 """The head over ``states`` answering ``targets``, choosing among ``penalties``. 

40 

41 Raises: 

42 UnsolvableHeadError: If there are fewer than two windows, which leaves nothing to 

43 leave out. 

44 """ 

45 x = states.detach().to("cpu").to(torch.float64) 

46 y = targets.detach().to("cpu").to(torch.float64) 

47 count = x.shape[0] 

48 if count < 2: 

49 raise UnsolvableHeadError( 

50 f"a head chosen by leave-one-out error needs at least two windows, got {count}" 

51 ) 

52 spread = x.std(dim=0, correction=0) 

53 scale = torch.where(spread > 0.0, spread, torch.ones_like(spread)) 

54 scaled = x / scale 

55 x_mean, y_mean = scaled.mean(dim=0), y.mean() 

56 centred, residual_targets = scaled - x_mean, y - y_mean 

57 left, singular, right_t = torch.linalg.svd(centred, full_matrices=False) 

58 projected = left.T @ residual_targets 

59 squared = singular**2 

60 errors: list[float] = [] 

61 coefficients: list[Tensor] = [] 

62 for penalty in penalties.values: 

63 shrink = squared / (squared + penalty) 

64 fitted = left @ (shrink * projected) 

65 # The intercept is unpenalised, so its leverage of one in ``count`` stays whole. 

66 leverage = (left**2 * shrink).sum(dim=1) + 1.0 / count 

67 loo = (residual_targets - fitted) / (1.0 - leverage) 

68 errors.append(float((loo**2).mean())) 

69 coefficients.append(right_t.T @ (singular / (squared + penalty) * projected)) 

70 chosen = min(range(len(errors)), key=lambda index: (errors[index], index)) 

71 coefficient = coefficients[chosen] 

72 return cls( 

73 weights=coefficient / scale, 

74 intercept=float(y_mean - (x_mean * coefficient).sum()), 

75 penalty=penalties.values[chosen], 

76 leave_one_out=tuple(errors), 

77 ) 

78 

79 def applied_to(self, head: RegressionHead) -> None: 

80 """Write the solution into ``head``, in the head's precision on the head's device.""" 

81 with torch.no_grad(): 

82 head.linear.weight.copy_(self.weights.to(head.linear.weight).unsqueeze(0)) 

83 head.linear.bias.fill_(self.intercept)
