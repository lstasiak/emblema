9class MaskedReconstruction(nn.Module): 

10 """The self-supervised objective: hide part of a window, encode the rest, predict the hidden. 

11 

12 A hidden token is marked as padding for the encoder's call, which for a set encoder is the same 

13 as never observed, so the encoder meets no placeholder it would not meet at inference. The 

14 decoder puts a placeholder at every hidden position, labelled with the encoder's own channel and 

15 time encodings, so a channel hidden whole still trains its embedding. The prediction covers 

16 every position and ``ReconstructionLoss`` decides which count; the masks come from outside, so 

17 tests and diagnostics can hold them fixed. 

18 

19 Attributes: 

20 encoder: The backbone being pretrained; what remains when the objective is done. 

21 decoder: The part that is thrown away. 

22 """ 

23 

24 def __init__(self, encoder: SetEncoder, *, decoder_layers: int, dropout: float = 0.0) -> None: 

25 super().__init__() 

26 self.encoder = encoder 

27 self.decoder = ReconstructionDecoder( 

28 encoder.architecture, layers=decoder_layers, dropout=dropout 

29 )

31 def forward(self, batch: TokenTensors, masks: TokenMasks) -> Tensor: 

32 hidden = masks.hidden 

33 states = self.encoder( 

34 batch.features, 

35 batch.channel_ids, 

36 batch.timestamps, 

37 batch.timeless, 

38 batch.padding_mask | hidden, 

39 ) 

40 encodings = self.encoder.channel_embedding(batch.channel_ids) + self.encoder.time_encoding( 

41 batch.timestamps, batch.timeless 

42 ) 

43 return self.decoder(states, hidden, encodings, batch.padding_mask)
