2from numpy.typing import NDArray 

3 

4from emblema.shared.kernel.ordering import seeded_rank 

5

7class Draws: 

8 """The random numbers one part of a synthetic corpus is made of, addressed by what it is. 

9 

10 A unit is generated on its own, without the units before it, so its numbers cannot come from 

11 a stream that has to be advanced to reach them. Each draw is addressed instead: a seed and 

12 the names of what it belongs to — a layout, a unit, a channel — become the state the numbers 

13 come from, through the same rule the rest of the project turns a seed into an order with, so 

14 that a seed becomes randomness in one place and not two. 

15 

16 Only uniform doubles are taken from the bit generator; the normal draws are built from them 

17 here. The numbers therefore follow from the bit stream alone, rather than from whichever 

18 algorithm the array library currently uses for a distribution — and a corpus regenerated 

19 later is the same corpus. 

20 """ 

21 

22 def __init__(self, seed: int, *parts: object) -> None: 

23 state = int.from_bytes(seeded_rank(seed, *parts)) 

24 self._generator = np.random.Generator(np.random.Philox(state)) 

25 

26 def uniform(self, *shape: int) -> NDArray[np.float64]: 

27 """Draws from ``[0, 1)``.""" 

28 return self._generator.random(shape) 

29 

30 def normal(self, *shape: int) -> NDArray[np.float64]: 

31 """Draws from the standard normal, by the Box-Muller transform of uniform pairs.""" 

32 count = int(np.prod(shape)) 

33 first, second = self.uniform(2, (count + 1) // 2) 

34 # The transform takes a logarithm, which the open end of the draw has to supply: the 

35 # numbers are drawn from [0, 1), so it is the lower end that has to be moved. 

36 radius = np.sqrt(-2.0 * np.log(1.0 - first)) 

37 angle = 2.0 * np.pi * second 

38 pairs = np.stack((radius * np.cos(angle), radius * np.sin(angle)), axis=1) 

39 return pairs.reshape(-1)[:count].reshape(shape)
