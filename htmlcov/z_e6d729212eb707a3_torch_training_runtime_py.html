111class TorchTrainingRuntime: 

112 """Trains the objective in this process, on whatever device the machine has. 

113 

114 The loop is the ordinary one — a forward pass under autocast, gradients accumulated over as 

115 many micro-batches as the budget asks for, one optimiser step, one scheduler step — and what 

116 it adds is that it can be stopped and picked up, and that it keeps the epoch worth keeping. 

117 Checkpoints are written on a step boundary, when no gradient is pending, and carry the 

118 position so a resumed run skips exactly the micro-batches the interrupted one had consumed. 

119 Each corpus of the mixture is batched on its own and the batches are interleaved in the order 

120 the seed gives, in turns of as many batches as a step accumulates, so a batch pads to its own 

121 corpus's windows, a step is taken over one corpus, and the run weighs a corpus by its share 

122 of the steps. Every epoch scores each corpus's held-out side apart; an epoch whose mean 

123 relative loss over the corpora is the lowest so far writes its weights durably, and the 

124 backbone the run ends with is that epoch's, whatever the last epoch did. 

125 

126 Masks are drawn on the host and moved, as the batches are, so a run on any device hides the 

127 same tokens; the device is where the arithmetic happens, never where the randomness comes 

128 from. 

129 """

295 def _epochs(self, session: _Session) -> Iterator[EpochOutcome]: 

296 budget = session.configuration.budget 

297 batches = len(session.training) 

298 while session.position.epoch < budget.epochs: 

299 if session.position.batches >= batches: 

300 # The checkpoint was written on the last step of its epoch, so that epoch is 

301 # behind and the run that wrote it reported the outcome. 

302 session.position = session.position.next_epoch() 

303 continue 

304 epoch = session.position.epoch 

305 started = time.perf_counter() 

306 training, checkpoint = self._epoch(session, batches) 

307 validation, hidden_ratio = self._validate(session) 

308 outcome = EpochOutcome( 

309 epoch=epoch, 

310 training_loss=training, 

311 validation=validation, 

312 hidden_ratio=hidden_ratio, 

313 seconds=time.perf_counter() - started, 

314 checkpoint=checkpoint, 

315 ) 

316 best = session.best 

317 if best is None or outcome.relative_validation < best.relative: 

318 best = _Best(outcome.relative_validation, self._write_model(session)) 

319 session.best = best 

320 outcome = replace(outcome, weights=best.weights) 

321 if epoch + 1 == budget.epochs: 

322 outcome = replace(outcome, backbone=best.weights) 

323 # The last micro-batch of an epoch always closes a step, so a checkpoint the policy 

324 # makes due on that step is written here, once the epoch is scored: it then carries 

325 # this epoch's best, and a run picked up from it, which starts at the next epoch, 

326 # keeps what this one kept. 

327 if session.configuration.checkpoint.due_at(session.position.steps): 

328 outcome = replace(outcome, checkpoint=self._write_checkpoint(session))

331 epoch + 1,

335 training, 

336 "; ".join( 

337 f"{scored.corpus} validation {scored.loss:.5f}, {scored.relative:.3f} of " 

338 f"the trivial predictor's" 

339 for scored in validation 

340 ), 

341 ) 

342 yield outcome 

343 session.position = session.position.next_epoch()

345 def _epoch(self, session: _Session, batches: int) -> tuple[float, ArtifactRef | None]: 

346 """One pass over the training windows from where the run stands. 

347 

348 Returns the error over its hidden tokens and the last checkpoint written during it — 

349 which is what a run interrupted in the next epoch would be picked up from. A checkpoint 

350 due on the epoch's last step is not written here but once the epoch has been scored. 

351 """ 

352 budget = session.configuration.budget 

353 skip = session.position.batches 

354 checkpoint: ArtifactRef | None = None 

355 session.model.train() 

356 epoch_error, epoch_tokens, group_tokens = 0.0, 0, 0 

357 progress = _Progress.at(session, batches) 

358 for index, on_host in enumerate(session.training.batches_of(session.position.epoch)): 

359 if index < skip: 

360 continue 

361 masks = session.masking.draw(on_host, session.draws) 

362 batch, drawn = on_host.to(self._device), masks.to(self._device) 

363 with session.precision.autocast(): 

364 error, scored = session.loss.summed( 

365 session.model(batch, drawn), batch, drawn.hidden 

366 ) 

367 summed = float(error.detach()) 

368 if not math.isfinite(summed): 

369 raise DivergedRunError( 

370 f"the loss of batch {index} in epoch {session.position.epoch} is {summed}, " 

371 f"after {session.position.steps} steps" 

372 ) 

373 progress.error += summed 

374 # Accumulated micro-batches make one gradient, and the error is summed rather than 

375 # averaged per batch so that each hidden token weighs the same whichever batch it 

376 # landed in. The sum is divided out of the gradients once, below, before the step. 

377 session.scaler.scale(error).backward() 

378 hidden = int(scored) 

379 group_tokens += hidden 

380 epoch_error += summed 

381 epoch_tokens += hidden 

382 progress.tokens += hidden 

383 stepped = budget.takes_a_step_at(index, batches) 

384 if stepped: 

385 self._average_gradients(session, group_tokens) 

386 # A gradient scaler skips a step whose gradients overflowed, which is what it is 

387 # for; without one, such a step would write non-finite weights into the model and 

388 # into the next checkpoint, so it is refused here instead. 

389 if not session.precision.scales_gradients: 389 ↛ 391line 389 didn't jump to line 391 because the condition on line 389 was always true

390 self._refuse_non_finite_gradients(session) 

391 session.scaler.step(session.optimiser) 

392 session.scaler.update() 

393 session.optimiser.zero_grad(set_to_none=True) 

394 session.scheduler.step() 

395 group_tokens = 0 

396 session.position = session.position.after_batch(stepped=stepped) 

397 if stepped: 

398 progress.stepped += 1 

399 if self._progress_every and progress.stepped % self._progress_every == 0: 

400 self._say_progress(session, progress) 

401 if ( 

402 stepped 

403 and index + 1 < batches 

404 and session.configuration.checkpoint.due_at(session.position.steps) 

405 ): 

406 checkpoint = self._write_checkpoint(session) 

407 return epoch_error / max(epoch_tokens, 1), checkpoint

449 def _validate(self, session: _Session) -> tuple[tuple[CorpusValidation, ...], float]: 

450 """What each corpus's held-out side cost this epoch, and the share of tokens scored. 

451 

452 The masks are drawn per batch from a generator the seed fixes, so every epoch is scored 

453 under the same hidden tokens: what moves between epochs is the model. The channel-mean 

454 predictor is scored over the same tokens in the same pass, so that what a corpus is worth 

455 against nothing learnt is measured rather than looked up. Each corpus is scored on its 

456 own loader and reported on its own: a single number over the mixture would be the 

457 largest corpus's. 

458 """ 

459 session.model.eval() 

460 scored_corpora: list[CorpusValidation] = [] 

461 hidden_tokens_in_all, observed_in_all = 0, 0 

462 with torch.no_grad(): 

463 for corpus, loader in session.validation: 

464 total, trivial, hidden_tokens = 0.0, 0.0, 0 

465 for index, on_host in enumerate(loader.batches_of(0)): 

466 draws = torch.Generator().manual_seed( 

467 session.configuration.budget.seed * VALIDATION_SEED_STRIDE + index 

468 ) 

469 masks = session.masking.draw(on_host, draws) 

470 batch, drawn = on_host.to(self._device), masks.to(self._device) 

471 with session.precision.autocast(): 

472 predicted = session.model(batch, drawn) 

473 error, counted = session.loss.summed(predicted, batch, drawn.hidden) 

474 nothing_learnt, _ = session.loss.summed_over_mean(batch, drawn.hidden) 

475 scored = int(counted) 

476 total += float(error) 

477 trivial += float(nothing_learnt) 

478 hidden_tokens += scored 

479 observed_in_all += int((~batch.padding_mask).sum()) 

480 hidden_tokens_in_all += hidden_tokens 

481 scored_corpora.append( 

482 CorpusValidation( 

483 corpus=corpus.name, 

484 tokens=hidden_tokens, 

485 loss=total / max(hidden_tokens, 1), 

486 trivial=trivial / max(hidden_tokens, 1), 

487 ) 

488 ) 

489 return tuple(scored_corpora), hidden_tokens_in_all / max(observed_in_all, 1)
