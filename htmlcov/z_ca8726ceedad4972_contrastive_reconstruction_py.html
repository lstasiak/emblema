20class ContrastiveReconstruction(MaskedReconstruction): 

21 """Masked reconstruction with a second term: the two halves of a window told apart from others'. 

22 

23 The masks cut every window in two. The tokens the encoder sees are one view and the tokens 

24 hidden from it are the other, so the second view is the first's complement, drawn by the same 

25 draw and no other: the reconstruction predicts the values of the second view from the first, 

26 and the contrast asks that each view, summarised to one point, pick the other view of its 

27 window out of every view of every other window in the batch. Two disjoint sets of 

28 observations of one stay or one month agree on what the window is, and a line through a 

29 token's neighbours does not answer that. 

30 

31 Each view goes through the encoder as the reconstruction's view does, the other view's 

32 tokens marked as padding; the first pass is the reconstruction's, so the decoder predicts 

33 from the same states the contrast pools. The projection is thrown away with the decoder. 

34 A window whose masks hid nothing of it has no second view and leaves the contrast, as a 

35 distractor too; it is still reconstructed, which is nothing. The encoder and the decoder are 

36 built before the projection, so a run under this objective draws the same encoder and the 

37 same decoder from its seed as the run that predicts values alone, and differs from it in the 

38 term only. 

39 

40 The contrast is weighed against the hidden tokens' loss as the term states: an anchor's mean 

41 loss, times the weight, times the hidden tokens scored, so that once the batch is divided by 

42 its tokens the loss is the mean reconstruction plus the weight times the mean contrast, and 

43 a micro-batch weighs by its tokens as it does under the reconstruction alone. The trivial 

44 predictor is the reconstruction's beside the contrast's chance, scaled the same way. 

45 

46 Attributes: 

47 encoder: The backbone being pretrained; what remains when the objective is done. 

48 decoder: The part that predicts the values; thrown away. 

49 projection: The part that carries a view's pooled state to the contrast; thrown away. 

50 term: The contrast's weight and temperature. 

51 """ 

52 

53 def __init__( 

54 self, 

55 encoder: SetEncoder, 

56 *, 

57 decoder_layers: int, 

58 contrast: ContrastiveTerm, 

59 dropout: float = 0.0, 

60 ) -> None: 

61 super().__init__(encoder, decoder_layers=decoder_layers, dropout=dropout) 

62 if not contrast.is_stated: 

63 raise ValueError("this objective contrasts the views of a window, which needs a term") 

64 self.projection = ContrastProjection(encoder.architecture) 

65 self.term = contrast

67 @staticmethod 

68 def views(batch: TokenTensors, masks: TokenMasks) -> tuple[Tensor, Tensor, Tensor]: 

69 """The padding of each view and which windows are pairs: ``(first, second, valid)``. 

70 

71 The first view's padding is the reconstruction's: the batch's own and the hidden tokens. 

72 The second's is everything else, so it holds exactly the hidden tokens that were 

73 observed. A window is a pair where each view holds a token. 

74 """ 

75 hidden = masks.hidden & ~batch.padding_mask 

76 first = batch.padding_mask | hidden 

77 second = ~hidden 

78 valid = (~first).any(dim=1) & hidden.any(dim=1) 

79 return first, second, valid 

80 

81 def contrasted(self, batch: TokenTensors, masks: TokenMasks) -> tuple[Tensor, Tensor]: 

82 """The projections of both views, ``[batch, width]`` each, read without the decoder.""" 

83 first, second, _ = self.views(batch, masks) 

84 return ( 

85 self.projection(self.encoded(batch, first), first), 

86 self.projection(self.encoded(batch, second), second), 

87 ) 

88 

89 def score(self, batch: TokenTensors, masks: TokenMasks, loss: ObjectiveLoss) -> ObjectiveScore: 

90 """The reconstruction's error and the weighed contrast, each beside the trivial cost.""" 

91 first, second, valid = self.views(batch, masks) 

92 states = self.encoded(batch, first) 

93 scorer = ReconstructionLoss(loss) 

94 prediction = self.decoded(states, batch, masks) 

95 reconstruction, tokens = scorer.summed(prediction, batch, masks.hidden) 

96 reconstruction_trivial, _ = scorer.summed_over_mean(batch, masks.hidden) 

97 

98 contrasting = ContrastiveLoss(self.term) 

99 contrast_sum, anchors = contrasting.summed( 

100 self.projection(states, first), 

101 self.projection(self.encoded(batch, second), second), 

102 valid, 

103 ) 

104 scale = self.term.weight * tokens.to(contrast_sum.dtype) / anchors.clamp(min=1) 

105 contrast = contrast_sum * scale 

106 contrast_trivial = contrasting.chance(valid) * scale 

107 

108 return ObjectiveScore( 

109 error=reconstruction + contrast, 

110 trivial=(reconstruction_trivial + contrast_trivial).detach(), 

111 tokens=tokens, 

112 terms={RECONSTRUCTION: reconstruction.detach(), CONTRAST: contrast.detach()}, 

113 trivial_terms={ 

114 RECONSTRUCTION: reconstruction_trivial.detach(), 

115 CONTRAST: contrast_trivial.detach(), 

116 }, 

117 )
