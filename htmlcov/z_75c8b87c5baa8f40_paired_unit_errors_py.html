11@dataclass(frozen=True, kw_only=True) 

12class PairedUnitErrors: 

13 """Two candidates' errors over the same units, unit by unit, so their difference is paired. 

14 

15 The unit is what a comparison is paired on and what an interval resamples: the error of a 

16 control and of a candidate are held per unit in one order, and any group of units — the whole 

17 side, or a resample of it — scores each side by adding sums and counts before a root is 

18 taken. Repeats of a cell pool here as well: the squared errors of every repeat are added per 

19 unit, so a cell's error is the root of the mean squared error over its repeats, and the 

20 spread between repeats is reported beside the interval rather than folded into it. 

21 

22 Invariants: at least one unit; both sides name the same units in the same order and count the 

23 same windows for each, since a pair over different windows is not a pair. 

24 

25 Attributes: 

26 control: The error per unit of what the candidate is compared against. 

27 candidate: The error per unit of what is compared. 

28 """ 

29 

30 control: tuple[UnitError, ...] 

31 candidate: tuple[UnitError, ...] 

32 

33 def __post_init__(self) -> None: 

34 if not self.control: 

35 raise InvalidPairedUnitErrorsError("a paired comparison needs at least one unit") 

36 if len(self.control) != len(self.candidate): 

37 raise InvalidPairedUnitErrorsError( 

38 f"{len(self.control)} control units paired with {len(self.candidate)} candidate " 

39 "units" 

40 ) 

41 for left, right in zip(self.control, self.candidate, strict=True): 

42 if left.unit != right.unit: 

43 raise InvalidPairedUnitErrorsError( 

44 f"control unit {left.unit} paired with candidate unit {right.unit}" 

45 ) 

46 if left.windows != right.windows: 

47 raise InvalidPairedUnitErrorsError( 

48 f"unit {left.unit} scored over {left.windows} control windows and " 

49 f"{right.windows} candidate windows" 

50 ) 

51 if len({error.unit for error in self.control}) != len(self.control): 

52 raise InvalidPairedUnitErrorsError("a unit is paired twice")

54 @classmethod 

55 def pooled( 

56 cls, 

57 control: Sequence[Sequence[UnitError]], 

58 candidate: Sequence[Sequence[UnitError]], 

59 ) -> Self: 

60 """One pair out of the repeats of both sides, the errors added per unit across repeats. 

61 

62 Each repeat is one run's error per unit; the repeats of a side are summed unit by unit, 

63 so a cell measured under several seeds is scored as one candidate over all of its answers. 

64 

65 Raises: 

66 InvalidPairedUnitErrorsError: If a side has no repeat, its repeats disagree about 

67 the units, or the two sides do not pair. 

68 """ 

69 return cls(control=_summed(control, "control"), candidate=_summed(candidate, "candidate"))

111def _rmse_over(errors: Sequence[UnitError], picks: Sequence[int]) -> float: 

112 squared = sum(errors[pick].squared_error for pick in picks) 

113 windows = sum(errors[pick].windows for pick in picks) 

114 return sqrt(squared / windows)

117def _summed(repeats: Sequence[Sequence[UnitError]], side: str) -> tuple[UnitError, ...]: 

118 if not repeats: 

119 raise InvalidPairedUnitErrorsError(f"the {side} side has no repeat to pool") 

120 first = tuple(repeats[0]) 

121 units = tuple(error.unit for error in first) 

122 squared = [error.squared_error for error in first] 

123 windows = [error.windows for error in first] 

124 for repeat in repeats[1:]: 

125 if tuple(error.unit for error in repeat) != units: 

126 raise InvalidPairedUnitErrorsError( 

127 f"the repeats of the {side} side disagree about the units" 

128 ) 

129 for index, error in enumerate(repeat): 

130 squared[index] += error.squared_error 

131 windows[index] += error.windows 

132 return tuple( 

133 UnitError(unit=unit, squared_error=total, windows=count) 

134 for unit, total, count in zip(units, squared, windows, strict=True) 

135 )
