12class HandoffTrainingRuntime: 

13 """A run made on a machine this one only handed work to, replayed from what it reported. 

14 

15 The third way to fulfil the port: not training on a device here and not counting epochs in 

16 memory, but reading what another machine trained and reporting it epoch by epoch as though 

17 it were happening now. The result is held to the run asked for — the configuration, the data 

18 and the checkpoint, each named when it differs — and to the store, which must hold the 

19 weights the result names. What passes is what the caller would have seen had the run 

20 happened here, so the same use case records it and the same tracker keeps the curve. 

21 

22 Refusals happen when the run is asked for, not when its first epoch is: a caller that 

23 records runs must not open one for a result that is not the run it ordered. 

24 """ 

25 

26 def __init__( 

27 self, exchange: HandoffExchange, store: ArtifactStore, result: ArtifactRef 

28 ) -> None: 

29 """Replay the result stored under ``result``, whose weights ``store`` must hold.""" 

30 self._exchange = exchange 

31 self._store = store 

32 self._result = result 

33 

34 def train( 

35 self, 

36 configuration: ExperimentConfiguration, 

37 mixture: TrainingMixture, 

38 resume_from: ArtifactRef | None = None, 

39 ) -> Iterator[EpochOutcome]: 

40 """The epochs the other machine reported, once the result is held to this run. 

41 

42 Raises: 

43 PretrainingResultRejectedError: If the result is of another configuration, other 

44 data or another checkpoint, or names weights the store does not hold. 

45 UnreadableHandoffDocumentError: If what is stored is not a result. 

46 ArtifactNotFoundError: If the result is not in the store. 

47 """ 

48 result = self._exchange.read_result(self._result) 

49 result.require_run_of(configuration, mixture.shape, resume_from) 

50 if not self._store.exists(result.outcome.backbone): 

51 raise PretrainingResultRejectedError( 

52 f"the result names weights under {result.outcome.backbone.key!r} " 

53 f"that the store does not hold" 

54 ) 

55 return iter(result.outcome.epochs)
