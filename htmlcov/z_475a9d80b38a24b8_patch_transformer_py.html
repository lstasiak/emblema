9class PatchTransformer(nn.Module): 

10 """A transformer over patches of each channel's row of the grid, as PatchTST reads a window. 

11 

12 Every channel is read on its own by the same encoder: its row is padded at the end by one 

13 stride, cut into overlapping patches, and each patch of values beside the same patch of its 

14 mask becomes one token. The mask travels with the values because the grid carried readings 

15 forward, and a token that could not tell a reading from a copy of the last one would learn 

16 from values the sensor never reported. A token's place in the window is a learnt embedding. 

17 

18 Unlike PatchTST no window is normalised by its own mean and spread: the level of a reading is 

19 what a remaining life is read from, and the values are already on the corpus's scale. 

20 

21 The channels meet only at the head: each channel's tokens are averaged into one state, the 

22 states are laid side by side, and a linear map reads one number out of them. So the head is 

23 as wide as the corpus's task has channels, and a model trained on one layout reads no other. 

24 """ 

25 

26 def __init__( 

27 self, spec: PatchModelSpec, *, channels: int, steps: int, starting_at: float 

28 ) -> None: 

29 """A model over rows of ``steps`` steps for ``channels`` channels. 

30 

31 Args: 

32 spec: How rows are cut into patches and how large the encoder is. 

33 channels: How many channels a window is read over. 

34 steps: How many steps each row of the grid has. 

35 starting_at: What the head answers before any step, in units of the label ceiling. 

36 

37 Raises: 

38 InvalidPatchModelSpecError: If a row is shorter than one patch. 

39 """ 

40 super().__init__() 

41 self._patch_length = spec.patch_length 

42 self._stride = spec.stride 

43 self.embedding = nn.Linear(2 * spec.patch_length, spec.width) 

44 self.position = nn.Parameter(torch.randn(spec.patches_over(steps), spec.width) * 0.02) 

45 self.dropout = nn.Dropout(spec.dropout) 

46 self.encoder = nn.TransformerEncoder( 

47 nn.TransformerEncoderLayer( 

48 spec.width, 

49 spec.heads, 

50 spec.feedforward_width, 

51 spec.dropout, 

52 activation="gelu", 

53 batch_first=True, 

54 norm_first=True, 

55 ), 

56 spec.layers, 

57 norm=nn.LayerNorm(spec.width), 

58 # The fast path for nested tensors needs post-norm layers; these are pre-norm. 

59 enable_nested_tensor=False, 

60 ) 

61 self.head = RegressionHead(channels * spec.width, starting_at=starting_at) 

62 

63 def forward(self, values: Tensor, observed: Tensor) -> Tensor: 

64 """One answer per window out of its grid, ``[batch, channels, steps]`` twice. 

65 

66 Args: 

67 values: What the grid holds per channel and step. 

68 observed: One where a reading was really made, zero where it was carried. 

69 """ 

70 batch, channels, _ = values.shape 

71 patches = torch.cat((self._patched(values), self._patched(observed)), dim=-1) 

72 tokens = self.dropout(self.embedding(patches) + self.position) 

73 encoded = self.encoder(tokens.reshape(batch * channels, -1, tokens.shape[-1])) 

74 states = encoded.mean(dim=1).reshape(batch, -1) 

75 answer: Tensor = self.head(states) 

76 return answer 

77 

78 def _patched(self, rows: Tensor) -> Tensor: 

79 """Each row padded by one stride of its last step, then cut into patches.""" 

80 padded = pad(rows, (0, self._stride), mode="replicate") 

81 return padded.unfold(-1, self._patch_length, self._stride)
