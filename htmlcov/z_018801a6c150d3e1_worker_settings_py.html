11class WorkerSettings(BaseModel): 

12 """What the campaign worker is told before it starts, read as ``EMBLEMA_WORKER__<FIELD>``. 

13 

14 A worker serves one backbone and runs every cell under one schedule, which is why both are 

15 told to the process rather than carried by each job. Only the device has a default, because 

16 hardware is the one thing an experiment does not declare. 

17 """ 

18 

19 workspace: Path = Field(description="Directory corpus blocks are fetched to and mapped from.") 

20 corpora: Path = Field(description="Directory the raw corpora sit in, where labels are read.")

22 schedule: ScheduleSettings

28 def backbone_ref(self) -> ArtifactRef: 

29 """The backbone as the registry holds it: a key and the checksum of its bytes. 

30 

31 Raises: 

32 ValueError: If the text is not a reference of the form ``key@algorithm:digest``. 

33 """ 

34 key, _, checksum = self.backbone.partition("@") 

35 algorithm, _, digest = checksum.partition(":") 

36 if not key or not algorithm or not digest: 

37 raise ValueError( 

38 f"not an artifact reference of the form key@algorithm:digest: {self.backbone!r}" 

39 ) 

40 return ArtifactRef(key, Checksum(HashAlgorithm(algorithm), digest))
