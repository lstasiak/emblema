14class WorkerSettings(BaseModel): 

15 """What a campaign worker is told before it starts, read as ``EMBLEMA_WORKER__<FIELD>``. 

16 

17 A worker serves one backbone and runs every cell under one schedule, which is why both are 

18 told to the process rather than carried by each job. Only the device has a default, because 

19 hardware is the one thing an experiment does not declare. 

20 

21 What a worker needs depends on what it competes. The one that trains networks has no use for 

22 the classical knobs and the one that fits baselines has no use for a backbone, a schedule, 

23 low-rank updates or a patch model's shape, so each of them is optional here and demanded by 

24 the process that cannot run without it. Demanded as it is assembled, so a worker told to 

25 serve something it was not configured for stops at startup rather than at the first cell it 

26 is handed. 

27 """ 

28 

29 workspace: Path = Field(description="Directory corpus blocks are fetched to and mapped from.") 

30 corpora: Path = Field(description="Directory the raw corpora sit in, where labels are read.") 

31 backbone: str | None = Field( 

32 default=None, description="Written key@algorithm:digest, as the registry holds it." 

33 ) 

34 schedule: ScheduleSettings | None = None 

35 lora: LoraSettings | None = None 

36 boosting: BoostingSettings | None = None 

37 convolutions: ConvolutionSettings | None = None 

38 patch: PatchSettings | None = None 

39 device: str | None = Field( 

40 default=None, description="Where a cell computes; the machine's accelerator unless given." 

41 )

93 def require_backbone_ref(self) -> ArtifactRef: 

94 """The backbone as the registry holds it: a key and the checksum of its bytes. 

95 

96 Raises: 

97 ValueError: If nothing is configured, or the text is not a reference of the form 

98 ``key@algorithm:digest``. 

99 """ 

100 if self.backbone is None: 

101 raise ValueError("this worker adapts a backbone and was given none to serve") 

102 key, _, checksum = self.backbone.partition("@") 

103 algorithm, _, digest = checksum.partition(":") 

104 if not key or not algorithm or not digest: 

105 raise ValueError( 

106 f"not an artifact reference of the form key@algorithm:digest: {self.backbone!r}" 

107 ) 

108 return ArtifactRef(key, Checksum(HashAlgorithm(algorithm), digest))
