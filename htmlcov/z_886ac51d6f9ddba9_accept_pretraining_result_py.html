28class AcceptPretrainingResult: 

29 """Takes a result back from the machine that trained and registers the weights it delivered. 

30 

31 The result is held to the backbone in the registry — which backbone, which code, which 

32 configuration, which data, which run — and then replayed through the same use case any run 

33 goes through, with the runtime that reads results: the tracker records it as it would a run 

34 made here, and the runtime holds the epochs to the configuration and the corpus this process 

35 reads for itself. What the runtime replayed is then held to the result accepted, because the 

36 runtime was built on a result of its own and nothing but that comparison says the two are 

37 one document. Only then is the backbone delivered, with the result beside the weights, and 

38 saved. A backbone that is already ready refuses before anything is replayed, so a delivery 

39 taken twice does not record a run twice. 

40 """ 

41

56 def __call__(self, command: AcceptPretrainingResultCommand) -> BackboneId: 

57 """Accept the result and return the backbone it made ready. 

58 

59 Raises: 

60 BackboneNotFoundError: If the result is for a backbone the registry does not know. 

61 BackboneAlreadyDeliveredError: If that backbone already has its weights. 

62 PretrainingResultRejectedError: If the result is not the delivery the backbone waits 

63 for, not the run this process reads the order as, or not the result the runtime 

64 replayed. 

65 UnreadableHandoffDocumentError: If what the reference names is not a result. 

66 """ 

67 result = self._exchange.read_result(command.result) 

68 backbone = self._backbones.get(result.backbone) 

69 backbone.require_open() 

70 result.require_delivery_for(backbone) 

71 mixture = TrainingMixture( 

72 corpora=tuple( 

73 self._reader.read(read.manifest, backbone.configuration.corpus_share) 

74 for read in backbone.inputs 

75 ) 

76 ) 

77 outcome = self._pretrain( 

78 PretrainBackboneCommand( 

79 configuration=backbone.configuration, 

80 mixture=mixture, 

81 run=backbone.run, 

82 resume_from=result.resumed_from, 

83 ) 

84 )
