5class FourierTimeEncoding(nn.Module): 

6 """A token's position in its window as sines and cosines at fixed frequencies. 

7 

8 The frequencies double from half a cycle per window upwards and are not learned: fixed, the 

9 encoding has nothing to overfit and extrapolates to positions training never showed. The sines 

10 and cosines are projected to the encoder's width. A timeless token gets a zero encoding rather 

11 than one at an invented time, chosen with ``where`` so that the module traces for export. 

12 

13 What is stored is the octave of each frequency rather than the frequency itself, and the angles 

14 are formed in single precision whatever precision the module is held in: a wide encoding reaches 

15 some 10^5 radians per window at its top frequency, which overflows half precision, while the 

16 octaves are small integers every floating type holds exactly. 

17 

18 Attributes: 

19 octaves: Doublings of half a cycle per window, one per frequency. 

20 """ 

21 

22 octaves: Tensor 

23 

24 def __init__(self, frequencies: int, width: int) -> None: 

25 super().__init__() 

26 self.register_buffer("octaves", torch.arange(frequencies, dtype=torch.float32)) 

27 self.projection = nn.Linear(2 * frequencies, width) 

28 

29 @property 

30 def frequencies(self) -> Tensor: 

31 """Angular frequencies, in radians per window, a position is multiplied by.""" 

32 return torch.pi * torch.exp2(self.octaves.float()) 

33 

34 def forward(self, timestamps: Tensor, timeless: Tensor) -> Tensor: 

35 angles = timestamps.unsqueeze(-1).float() * self.frequencies 

36 features = torch.cat([angles.sin(), angles.cos()], dim=-1) 

37 encoded = self.projection(features.to(self.projection.weight.dtype)) 

38 return torch.where(timeless.unsqueeze(-1), torch.zeros_like(encoded), encoded)
