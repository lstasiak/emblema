9@dataclass(frozen=True) 

10class OutcomeFolds: 

11 """Which windows of a sample a penalty is judged on, each by a head fitted without them. 

12 

13 A head over outcomes fitted by penalised likelihood has no leave-one-out error for free, as 

14 a least-squares one has, so its penalty is chosen by the log-loss of folds held out in 

15 turn. The folds keep the sample's proportion of the outcomes: a fold richer in positives 

16 than the rest would leave a head fitted on fewer of them to answer it. 

17 

18 Each outcome's windows are dealt to the folds in turn, in the sample's order, the second 

19 outcome taking up the turn where the first left off, so every fold holds each outcome within 

20 one window of its share and the folds differ in size by one at most. The order is the one 

21 the sample was drawn into, so the folds replay with it and need no seed of their own. 

22 

23 Invariants: at least one window; the folds are numbered from zero and each holds a window. 

24 

25 Attributes: 

26 assignment: The fold of each window, in the sample's order. 

27 """ 

28 

29 # The count cross-validation usually takes: each head is fitted on four fifths of the sample, 

30 # close to the head it stands in for, at the cost of five fits per penalty. 

31 COUNT: ClassVar[int] = 5 

32 # Two windows of an outcome, dealt in turn, land in two folds, so every head fitted without 

33 # a fold sees both outcomes. 

34 _FEWEST_OF_EACH: ClassVar[int] = 2 

35 

36 assignment: tuple[int, ...] 

37 

38 def __post_init__(self) -> None: 

39 folds = set(self.assignment) 

40 if not folds or folds != set(range(len(folds))): 

41 raise InvalidOutcomeFoldsError( 

42 f"folds must be numbered from zero, each holding a window, got {sorted(folds)}" 

43 ) 

44 

45 @classmethod 

46 def of(cls, outcomes: Sequence[float]) -> Self: 

47 """The folds of a sample whose windows have ``outcomes``, in the sample's order. 

48 

49 Raises: 

50 UnfoldableOutcomesError: If an outcome is not zero or one, or either outcome has 

51 fewer than two windows, which leaves a fold whose head never sees it. 

52 """ 

53 if any(outcome not in OutcomeScheme.OUTCOMES for outcome in outcomes): 

54 raise UnfoldableOutcomesError("every outcome must be zero or one") 

55 for outcome in OutcomeScheme.OUTCOMES: 

56 held = sum(1 for value in outcomes if value == outcome) 

57 if held < cls._FEWEST_OF_EACH: 

58 raise UnfoldableOutcomesError( 

59 f"folds need at least {cls._FEWEST_OF_EACH} windows of each outcome, " 

60 f"got {held} of outcome {outcome:g}" 

61 ) 

62 count = min(cls.COUNT, len(outcomes)) 

63 assignment = [0] * len(outcomes) 

64 turn = 0 

65 for outcome in OutcomeScheme.OUTCOMES: 

66 for place, value in enumerate(outcomes): 

67 if value == outcome: 

68 assignment[place] = turn % count 

69 turn += 1 

70 return cls(assignment=tuple(assignment)) 

71 

72 @property 

73 def count(self) -> int: 

74 return max(self.assignment) + 1 

75 

76 def held_out(self, fold: int) -> tuple[int, ...]: 

77 """Places of the windows ``fold`` is judged on.""" 

78 return tuple(place for place, owner in enumerate(self.assignment) if owner == fold) 

79 

80 def kept(self, fold: int) -> tuple[int, ...]: 

81 """Places of the windows the head judged on ``fold`` is fitted on.""" 

82 return tuple(place for place, owner in enumerate(self.assignment) if owner != fold)
