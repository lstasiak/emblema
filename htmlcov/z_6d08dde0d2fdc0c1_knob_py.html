1"""Turning one field of a specification to a value written as text, the way a variant names it. 

2 

3Every specification with knobs turns them the same way — the value is read as the type the 

4field already holds, and the specification then judges it — so the reading is stated once here 

5rather than once per method. A value that does not read as that type, or that the specification 

6refuses, is a knob the specification cannot be turned to. 

7""" 

8 

9from dataclasses import replace 

10from typing import TYPE_CHECKING, Any 

11 

12from emblema.evaluation.domain.exceptions import EvaluationError, UnknownKnobError 

13 

14if TYPE_CHECKING: 

15 from _typeshed import DataclassInstance 

16 

17 

18def turned[T: DataclassInstance](specification: T, field: str, value: str) -> T: 

19 """``specification`` with ``field`` set to ``value``, read as the type the field holds. 

20 

21 Bound to any dataclass rather than to the specifications that have knobs, because naming 

22 them here would make this package depend on each of theirs while they depend on it. 

23 

24 Raises: 

25 UnknownKnobError: If the value does not read as that type, or the specification 

26 refuses it. 

27 """ 

28 current: Any = getattr(specification, field) # a knob is an int or a float field 

29 try: 

30 read = type(current)(value) 

31 except ValueError as error: 

32 raise UnknownKnobError( 

33 f"{field} takes a {type(current).__name__}, got {value!r}" 

34 ) from error 

35 try: 

36 return replace(specification, **{field: read}) 

37 except (EvaluationError, ValueError) as error: 

38 raise UnknownKnobError(f"{field} cannot be {value}: {error}") from error
