2from torch.nn.functional import scaled_dot_product_attention

5class SelfAttention(nn.Module): 

6 """Full self-attention over the tokens of a window, with padding kept out of the keys. 

7 

8 Exact and quadratic in the window: the window length bounds the cost, not an approximation of 

9 the attention. ``scaled_dot_product_attention`` and nothing else, because it is the 

10 implementation that returns numbers rather than NaN on a window with no observed token, in 

11 PyTorch and in the exported graph alike. Padding positions are hidden as keys only; they still 

12 produce a state, which the consumer masks. 

13 """ 

14 

15 def __init__(self, width: int, heads: int, dropout: float) -> None: 

16 super().__init__() 

17 self.heads = heads 

18 self.dropout = dropout 

19 self.qkv = nn.Linear(width, 3 * width) 

20 self.projection = nn.Linear(width, width) 

21 

22 def forward(self, states: Tensor, padding_mask: Tensor) -> Tensor: 

23 query, key, value = self.qkv(states).chunk(3, dim=-1) 

24 keep = ~padding_mask[:, None, None, :] 

25 attended = scaled_dot_product_attention( 

26 self._split_heads(query), 

27 self._split_heads(key), 

28 self._split_heads(value), 

29 attn_mask=keep, 

30 dropout_p=self.dropout if self.training else 0.0, 

31 ) 

32 return self.projection(self._merge_heads(attended)) 

33 

34 def _split_heads(self, states: Tensor) -> Tensor: 

35 """``[batch, tokens, width]`` to ``[batch, heads, tokens, head width]``.""" 

36 batch, tokens, width = states.shape 

37 return states.reshape(batch, tokens, self.heads, width // self.heads).transpose(1, 2) 

38 

39 @staticmethod 

40 def _merge_heads(attended: Tensor) -> Tensor: 

41 batch, heads, tokens, head_width = attended.shape 

42 return attended.transpose(1, 2).reshape(batch, tokens, heads * head_width)
