18"""What a candidate gets wrong on each unit of a cell, in unit order."""

21class InMemoryCandidateProvider: 

22 """Answers with errors stated up front, so a whole campaign runs without arithmetic. 

23 

24 Nothing is learnt here: what each cell reports is decided by the function the provider was 

25 given, one error per unit. That is enough to exercise everything around the candidates — the 

26 grid expanding, cells recording, a campaign refusing to finish early, the comparison reading 

27 what ran — in the time it takes to call a function, and it is the only way to write those 

28 tests against known numbers rather than against whatever a model happened to produce. 

29 """ 

30 

31 def __init__( 

32 self, 

33 candidates: Sequence[CampaignCandidate], 

34 units: Sequence[UnitKey], 

35 errors: StatedErrors, 

36 store: ArtifactStore | None = None, 

37 ) -> None: 

38 """Answer for ``candidates`` over ``units``, reporting what ``errors`` says. 

39 

40 Args: 

41 candidates: Who this provider supplies. 

42 units: The units every cell is scored over. Held in name order, the one a real 

43 run reports its errors in, so that two candidates always pair. 

44 errors: What a cell gets wrong on each unit, one error per unit. 

45 store: Where a candidate this provider is asked to keep is put; a provider never 

46 asked to keep one needs none. 

47 """ 

48 self._candidates = tuple(candidates) 

49 self._units = tuple(sorted(units, key=str)) 

50 self._errors = errors 

51 self._store = store 

52 

53 def describe(self, candidate: CandidateRef) -> CampaignCandidate: 

54 for known in self._candidates: 

55 if known.ref == candidate: 

56 return known 

57 raise UnknownCandidateError(f"this provider supplies no candidate {candidate}") 

58 

59 def evaluate(self, request: CandidateEvaluation) -> CellResult: 

60 """Answer one cell with the errors this provider was told to report. 

61 

62 Raises: 

63 UnknownCandidateError: If this provider supplies no candidate of that name. 

64 CandidateMismatchError: If the campaign recorded the candidate as anything other 

65 than what this provider states for it. 

66 """ 

67 request.declared.must_match(self.describe(request.cell.candidate)) 

68 errors = self._errors(request.cell) 

69 return CellResult( 

70 cell=request.cell, 

71 errors=tuple( 

72 UnitError(unit=unit, squared_error=error**2, windows=1) 

73 for unit, error in zip(self._units, errors, strict=True) 

74 ), 

75 seconds=0.0, 

76 artifact=self._kept(request.cell) if request.retain else None, 

77 ) 

78 

79 def _kept(self, cell: CampaignCell) -> ArtifactRef: 

80 """The cell's answers, stored, so a retained cell names bytes that really exist. 

81 

82 Raises: 

83 CandidateNotRetainableError: If the provider was given nowhere to keep them. 

84 """ 

85 if self._store is None: 

86 raise CandidateNotRetainableError( 

87 "this provider was asked to keep what it fitted and was given no store" 

88 ) 

89 return self._store.put(repr(tuple(self._errors(cell))).encode())
