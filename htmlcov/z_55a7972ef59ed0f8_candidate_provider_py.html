68class InMemoryCandidateProvider: 

69 """Answers with results stated up front, so a whole campaign runs without arithmetic. 

70 

71 Nothing is learnt here: what each cell reports is decided by the function the provider was 

72 given, an error per unit or an answer per unit. That is enough to exercise everything around 

73 the candidates, from the grid expanding to the comparison reading what ran, in the time it 

74 takes to call a function, against known numbers rather than whatever a model produced. 

75 """ 

76 

77 def __init__( 

78 self, 

79 candidates: Sequence[CampaignCandidate], 

80 units: Sequence[UnitKey], 

81 stated: StatedErrors | StatedAnswers, 

82 store: ArtifactStore | None = None, 

83 ) -> None: 

84 """Answer for ``candidates`` over ``units``, reporting what ``stated`` says of each cell. 

85 

86 Args: 

87 candidates: Who this provider supplies. 

88 units: The units every cell is scored over. Held in name order, the one a real 

89 run reports its errors in, so that two candidates always pair. 

90 stated: What a cell scores: its errors per unit, or its answers. 

91 store: Where a candidate this provider is asked to keep is put; a provider never 

92 asked to keep one needs none. 

93 """ 

94 self._candidates = tuple(candidates) 

95 self._units = tuple(sorted(units, key=str)) 

96 self._stated = stated 

97 self._store = store

99 def describe(self, candidate: CandidateRef) -> CampaignCandidate: 

100 for known in self._candidates: 

101 if known.ref == candidate: 

102 return known 

103 raise UnknownCandidateError(f"this provider supplies no candidate {candidate}")

105 def evaluate(self, request: CandidateEvaluation) -> CellResult: 

106 """Answer one cell with what this provider was told to report for it. 

107 

108 Raises: 

109 UnknownCandidateError: If this provider supplies no candidate of that name. 

110 CandidateMismatchError: If the campaign recorded the candidate as anything other 

111 than what this provider states for it. 

112 """ 

113 request.declared.must_match(self.describe(request.cell.candidate)) 

114 errors, predictions = self._stated.scored(request.cell, self._units) 

115 return CellResult( 

116 cell=request.cell, 

117 errors=errors, 

118 seconds=0.0, 

119 artifact=self._kept(request.cell) if request.retain else None, 

120 predictions=predictions, 

121 )

123 def _kept(self, cell: CampaignCell) -> ArtifactRef: 

124 """What the cell was told to report, stored, so a retained cell names bytes that exist. 

125

129 if self._store is None:

132 ) 

133 return self._store.put(repr(tuple(self._stated.of(cell))).encode())
