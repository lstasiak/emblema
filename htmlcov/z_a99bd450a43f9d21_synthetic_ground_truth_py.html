13class SyntheticGroundTruth: 

14 """The exact reading of one sensor of a generated corpus, at the instant a task asks about. 

15 

16 The corpus is generated from its specification, so its truth is generated the same way: the 

17 signal a sensor would report without noise, at any instant, whether or not the sensor 

18 reported there. That is what the forecasting task of the synthetic control asks for, and 

19 what its label scheme names — which sensor, how far past a window's end — this adapter reads 

20 off the scheme, so the question is stated once. The reading is rounded to the precision of 

21 the corpus, as the sensor's own readings are. 

22 

23 Units are named ``<layout>/<index>``, as the reader of these corpora names them. 

24 """ 

25 

26 def __init__(self, signal: SensorSignal, scheme: ForecastScheme) -> None: 

27 """Read the truth of ``scheme`` off ``signal``. 

28 

29 Raises: 

30 UnknownGroundTruthError: If the layout has no sensor of the scheme's name. 

31 """ 

32 names = signal.layout.channel_names 

33 if scheme.channel not in names: 

34 raise UnknownGroundTruthError( 

35 f"layout {signal.layout.name!r} has no channel {scheme.channel!r}" 

36 ) 

37 self._signal = signal 

38 self._scheme = scheme 

39 self._channel = names.index(scheme.channel) 

40 

41 def truths_of(self, corpus: str, windows: Sequence[TaskWindow]) -> Mapping[TaskWindow, float]: 

42 # This adapter is the truth of one generated corpus; the name is read above it. 

43 by_unit: dict[int, list[TaskWindow]] = defaultdict(list) 

44 for window in windows: 

45 by_unit[self._index_of(window.unit)].append(window) 

46 truths: dict[TaskWindow, float] = {} 

47 for index, of_unit in by_unit.items(): 

48 instants = np.array([self._scheme.instant_of(w.ends_at) for w in of_unit]) 

49 values = self._signal.values_at(index, self._channel, instants) 

50 for window, value in zip(of_unit, values.tolist(), strict=True): 

51 truths[window] = round(float(value), PRECISION) 

52 return truths 

53 

54 def _index_of(self, unit: UnitKey) -> int: 

55 layout = self._signal.layout 

56 prefix, _, index = str(unit).partition("/") 

57 if ( 

58 prefix == layout.name 

59 and index.isascii() 

60 and index.isdigit() 

61 and str(int(index)) == index 

62 and int(index) < layout.units 

63 ): 

64 return int(index) 

65 raise UnknownGroundTruthError(f"{unit} is not a unit of layout {layout.name!r}")
