2 

3A function rather than a class, because there is no state to hold. Stated once because two 

4applications configured apart would agree until the day one of them gained an option. 

5"""

7from celery import Celery 

8 

9from emblema.shared.jobs.worker_pool import WorkerPool 

10 

11 

12def celery_application(broker_url: str, *, name: str = "emblema") -> Celery: 

13 """A Celery application on that broker, configured as every process of this system needs it. 

14 

15 Late acknowledgement, so a cell whose worker died is delivered again rather than lost; the 

16 use case answers a cell it has already recorded from what is stored, so a second delivery 

17 costs a query. On AMQP that acknowledgement is the broker's own: a job stays unacknowledged 

18 for exactly as long as the worker holds it, however many hours a cell takes, with no timeout 

19 to guess at. Publishing waits for the broker to confirm, so a submission that returns is one 

20 the broker took responsibility for rather than one that was written to a socket. 

21 """ 

22 app = Celery(name, broker=broker_url) 

23 app.conf.task_ignore_result = True 

24 app.conf.task_acks_late = True 

25 app.conf.task_default_queue = str(WorkerPool.GENERAL) 

26 app.conf.broker_transport_options = {"confirm_publish": True} 

27 # Celery's worker-to-worker mailbox declares transient non-exclusive queues, which RabbitMQ 4 

28 # refuses outright: a worker that opens one never finishes starting. Nothing here inspects or 

29 # controls workers at runtime, so it is switched off rather than the broker asked to permit 

30 # what it has deprecated. This covers the mailbox; the startup handshakes that also use it 

31 # are command-line flags with no setting behind them, so the worker is started with 

32 # `--without-mingle --without-gossip` (README). 

33 app.conf.worker_enable_remote_control = False 

34 return app
