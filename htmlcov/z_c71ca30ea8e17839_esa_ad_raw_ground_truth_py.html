28class EsaAdRawGroundTruth: 

29 """Which annotated event, if any, each window of the raw satellite telemetry overlaps. 

30 

31 The benchmark publishes, beside each mission, ``labels.csv`` with one row per channel an 

32 event touches, each with the event's identifier and the instants it starts and ends on that 

33 channel, and ``anomaly_types.csv`` with the category of each event. Only the events touching 

34 a channel the corpus reads count, since a window holds no other channel; of those, an 

35 anomaly and a rare nominal event are events alike, as the benchmark scores them, and a gap 

36 in the downlink is a stretch the task does not score. Events are numbered from one in the 

37 order the file lists them, among the events that count. A window answers with the lowest 

38 number of the events it overlaps, zero where it overlaps none, and the excluded mark where 

39 it overlaps a gap and no event. 

40 

41 A window ends at the hour the task names and starts a window's length before it, on the 

42 axis the reader of the corpus placed its observations on: hours since the stated origin. 

43 Units are named ``<mission>/<YYYY-MM>``, as the reader names them; a mission's files are 

44 read once and kept. 

45 """ 

46 

47 _LABELS: ClassVar[str] = "labels.csv" 

48 _TYPES: ClassVar[str] = "anomaly_types.csv" 

49 _GAP: ClassVar[str] = "Communication Gap"

51 def __init__( 

52 self, 

53 root: Path, 

54 *, 

55 channels: Mapping[str, Sequence[int]], 

56 origin_seconds: int, 

57 window_hours: float, 

58 ) -> None: 

59 """Answer for the missions in ``root``, over the channels each mission's corpus reads. 

60 

61 Args: 

62 root: The directory holding one directory per mission. 

63 channels: Channel numbers the corpus reads, per mission; events touching no such 

64 channel do not count. 

65 origin_seconds: The epoch second the corpus's hours are counted from. 

66 window_hours: How long each window is, so its start follows from its end. 

67 """ 

68 self._root = root 

69 self._channels = {mission: set(numbers) for mission, numbers in channels.items()} 

70 self._origin = origin_seconds 

71 self._window = window_hours 

72 self._events: dict[str, tuple[_Event, ...]] = {} 

73 

74 def truths_of(self, corpus: str, windows: Sequence[TaskWindow]) -> Mapping[TaskWindow, float]: 

75 # This adapter is the truth of one corpus, so the name is the composite's business. 

76 return {window: self._answer(window) for window in windows}

78 def _answer(self, window: TaskWindow) -> float: 

79 mission, separator, _ = str(window.unit).partition("/") 

80 if not separator or mission not in self._channels: 

81 raise UnknownGroundTruthError(f"{window.unit} is not a month of a mission read here") 

82 end = self._origin + window.ends_at * _SECONDS_PER_HOUR 

83 start = end - self._window * _SECONDS_PER_HOUR 

84 overlapping = [ 

85 event for event in self._of_mission(mission) if event.start < end and start < event.end 

86 ] 

87 numbered = [event.number for event in overlapping if event.scored] 

88 if numbered: 

89 return float(min(numbered)) 

90 if overlapping: 

91 return AnomalyScheme.EXCLUDED 

92 return AnomalyScheme.NORMAL 

93 

94 def _of_mission(self, mission: str) -> tuple[_Event, ...]: 

95 if mission not in self._events: 

96 self._events[mission] = self._read(mission) 

97 return self._events[mission]

99 def _read(self, mission: str) -> tuple[_Event, ...]: 

100 folder = self._root / mission 

101 categories = self._categories(folder / self._TYPES) 

102 spans: dict[str, tuple[float, float]] = {} 

103 order: list[str] = [] 

104 for number, row in enumerate(self._rows(folder / self._LABELS), start=2): 

105 identifier, channel = row.get("ID"), row.get("Channel") 

106 if not identifier or not channel or not channel.startswith("channel_"): 

107 raise UnreadableGroundTruthError(f"{folder / self._LABELS} line {number}: no event") 

108 try: 

109 selected = int(channel.removeprefix("channel_")) in self._channels[mission] 

110 start = self._instant(row["StartTime"]) 

111 end = self._instant(row["EndTime"]) 

112 except (KeyError, ValueError) as error: 

113 raise UnreadableGroundTruthError( 

114 f"{folder / self._LABELS} line {number}: {error}" 

115 ) from error 

116 if not selected: 

117 continue 

118 if identifier not in spans: 

119 order.append(identifier) 

120 spans[identifier] = (start, end) 

121 else: 

122 first, last = spans[identifier] 

123 spans[identifier] = (min(first, start), max(last, end)) 

124 events = [] 

125 counted = 0 

126 for identifier in order: 

127 if identifier not in categories: 

128 raise UnreadableGroundTruthError( 

129 f"{folder / self._TYPES} names no category for event {identifier}" 

130 ) 

131 scored = categories[identifier] != self._GAP 

132 if scored: 

133 counted += 1 

134 start, end = spans[identifier] 

135 events.append( 

136 _Event(number=counted if scored else 0, start=start, end=end, scored=scored) 

137 ) 

138 return tuple(events)

140 def _categories(self, path: Path) -> dict[str, str]: 

141 categories: dict[str, str] = {} 

142 for number, row in enumerate(self._rows(path), start=2): 

143 identifier, category = row.get("ID"), row.get("Category") 

144 if not identifier or category is None: 

145 raise UnreadableGroundTruthError(f"{path} line {number}: no event or category") 

146 categories[identifier] = category 

147 return categories
