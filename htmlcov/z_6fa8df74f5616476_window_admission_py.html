16class WindowAdmission: 

17 """Holds a request to what the model takes and the service allows, and tokenises it. 

18 

19 The one path from raw readings to tokens for both things a model answers, so a prediction 

20 and a representation of the same window are computed over the same tokens. The policy for 

21 a reading the model cannot take is the model input's; the tokenisation is the Catalog's; 

22 what is refused outright — too many windows, a window too long, a reading of the wrong 

23 kind — is the service's. A window shorter or longer than the candidate was fitted on is 

24 answered and remarked on: the arithmetic runs, and the caller is told the ground moved. 

25 

26 Every reading becomes one token, so a window too long is refused by counting its readings 

27 before any of them is tokenised: the tokeniser is the costly step, and a request that will 

28 be refused should not pay for it. 

29 """ 

30 

31 def __init__( 

32 self, runtime: InferenceRuntime, tokeniser: WindowTokeniser, limits: InferenceLimits 

33 ) -> None: 

34 self._runtime = runtime 

35 self._tokeniser = tokeniser 

36 self._limits = limits 

37 

38 def prepare( 

39 self, model: ServedModel, windows: Sequence[ObservedWindow] 

40 ) -> tuple[PreparedWindow, ...]: 

41 """Every window of the request as the model is fed it, in the order given. 

42 

43 Raises: 

44 EmptyRequestError: If there is no window. 

45 TooManyWindowsError: If there are more than the service admits at once. 

46 UnobservedWindowError: If a window holds no reading the model takes. 

47 UntokenisableRequestError: If a reading is of the wrong kind for its channel. 

48 WindowTooLongError: If a window holds more readings than the service tokenises. 

49 ArtifactUnavailableError: If the model's artifact is not in the store. 

50 UnreadableServedArtifactError: If it is not a kept candidate the runtime reads. 

51 """ 

52 self._limits.admit_count(len(windows)) 

53 model_input = self._runtime.describe(model.artifact) 

54 return tuple(self._prepared(model_input, window) for window in windows) 

55 

56 def _prepared(self, model_input: ModelInput, window: ObservedWindow) -> PreparedWindow: 

57 admitted = model_input.admit(window) 

58 self._limits.admit_length( 

59 len(admitted.window.observations) + len(admitted.window.static_features) 

60 ) 

61 try: 

62 tokens = self._tokeniser.tokenise( 

63 admitted.window, model_input.corpus, model_input.channels 

64 ) 

65 except UntokenisableWindowError as error: 

66 raise UntokenisableRequestError(str(error)) from error 

67 return PreparedWindow( 

68 admitted=admitted, tokens=tokens, warnings=self._warnings(model_input, admitted) 

69 ) 

70 

71 @staticmethod 

72 def _warnings(model_input: ModelInput, admitted: AdmittedWindow) -> tuple[str, ...]: 

73 warnings = [] 

74 if admitted.ignored: 

75 warnings.append( 

76 "readings on channels the model does not know were ignored: " 

77 + ", ".join(admitted.ignored) 

78 ) 

79 length = admitted.window.length 

80 if not isclose(length, model_input.window_length): 

81 warnings.append( 

82 f"the window spans {length:g} and the candidate was fitted on windows of " 

83 f"{model_input.window_length:g}" 

84 ) 

85 return tuple(warnings)
