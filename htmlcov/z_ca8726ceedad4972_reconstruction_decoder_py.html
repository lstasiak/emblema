2from torch import Tensor, nn

8class ReconstructionDecoder(nn.Module): 

9 """Predicts the value of every token of a window from the states of the tokens it can see. 

10 

11 The encoder only ever saw the visible tokens; here the hidden ones re-enter as one learned 

12 vector each, told apart solely by the channel and time encodings added to every position, and 

13 a few blocks of the same kind the encoder is made of let them read the visible states. The 

14 head is a single linear map to the value. Everything here is discarded after pretraining: the 

15 backbone is the encoder, and nothing the encoder exports passes through this module. 

16 

17 The blocks attend with the batch's own padding mask, not the encoder's: a hidden token is a 

18 token the encoder pretended was absent, but it is present to the decoder, which has to 

19 predict it. 

20 """ 

21 

22 def __init__( 

23 self, architecture: EncoderArchitecture, *, layers: int, dropout: float = 0.0 

24 ) -> None: 

25 super().__init__() 

26 if layers < 1: 

27 raise ValueError( 

28 f"the decoder needs at least one block to reach the visible tokens, got {layers}" 

29 ) 

30 self.mask_token = nn.Parameter(torch.empty(architecture.width)) 

31 nn.init.normal_(self.mask_token, std=0.02) 

32 self.blocks = nn.ModuleList(EncoderBlock(architecture, dropout) for _ in range(layers)) 

33 self.norm = nn.LayerNorm(architecture.width) 

34 self.head = nn.Linear(architecture.width, 1) 

35 

36 def forward( 

37 self, states: Tensor, hidden: Tensor, encodings: Tensor, padding_mask: Tensor 

38 ) -> Tensor: 

39 placeholder = self.mask_token.to(states.dtype).expand_as(states) 

40 stream = torch.where(hidden.unsqueeze(-1), placeholder, states) + encodings 

41 for block in self.blocks: 

42 stream = block(stream, padding_mask) 

43 return self.head(self.norm(stream)).squeeze(-1)
