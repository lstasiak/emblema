10class ShortUnits(StrEnum): 

11 """What a unit shorter than one window yields. 

12 

13 A window is laid only where it fits, so a unit shorter than the window yields nothing and 

14 is silent in the published corpus. Where the units are records of varying length and their 

15 length is part of what they say, such as telemetry fragments cut by their operators, the 

16 unit yields one window of the nominal length anchored at its start, so every unit speaks and 

17 its duration is read off how much of the window it fills. 

18 

19 Attributes: 

20 DROPPED: A unit shorter than a window yields no window. 

21 ONE_WINDOW: A unit shorter than a window yields one window of the nominal length, 

22 starting with the unit; the unit's readings fill its beginning. 

23 """ 

24 

25 DROPPED = "dropped" 

26 ONE_WINDOW = "one_window" 

27

29@dataclass(frozen=True) 

30class WindowSpec: 

31 """How a unit's time axis is cut into windows: a length and a stride, in the unit's time unit. 

32 

33 Windows are laid from the start of a unit's extent, every ``stride``, and only a window that 

34 fits inside the extent counts; the tail that no full window covers is left out, and what a 

35 unit too short for one window yields is the policy's business. Both numbers are elapsed 

36 time, never a count of samples, so the same rule serves a corpus sampled every cycle and one 

37 sampled whenever a sensor reports. Scale lives in configuration: there is no default window. 

38 

39 Attributes: 

40 length: Extent of one window; positive, finite. 

41 stride: Distance between the starts of consecutive windows; positive, finite. 

42 short_units: What a unit shorter than one window yields. 

43 """ 

44 

45 length: float 

46 stride: float 

47 short_units: ShortUnits = ShortUnits.DROPPED 

48

51 raise InvalidWindowSpecError(f"length must be positive and finite, got {self.length}")

53 raise InvalidWindowSpecError(f"stride must be positive and finite, got {self.stride}")

55 def windows_over(self, extent: TimeExtent) -> Iterator[TimeExtent]: 

56 """Every window that fits inside ``extent``, in time order. 

57 

58 A unit shorter than one window yields one window of the nominal length under the policy 

59 that says so, which then reaches past the unit's end; its readings lie in its beginning. 

60 """ 

61 if extent.length < self.length: 

62 if self.short_units is ShortUnits.ONE_WINDOW: 

63 yield TimeExtent(extent.start, extent.start + self.length) 

64 return 

65 index = 0 

66 while True: 

67 start = extent.start + index * self.stride 

68 end = start + self.length 

69 if end > extent.end: 

70 return 

71 yield TimeExtent(start, end) 

72 index += 1
