1"""What a saturation curve says about its corpus, and the rules that read it. 

2 

3Three readings were fixed before any curve was measured, in the order a failure invalidates the 

4next: a model that memorises at the largest share says nothing about the data beyond that there 

5is too little of it for the model; a plateau says the corpus has been learnt as far as a model of 

6this size trained this long can learn it; a loss still falling says the data, not the model, is 

7the limit. A fourth was added after the first measurement, when the satellite corpus showed a 

8case the three had not foreseen — a held-out side no run does better on than the channel mean, 

9which the rules read as overfitting — and comes before them all: nothing learnt is a precondition 

10of the other three, not a threshold moved. The thresholds are conventions and are changed by 

11editing them, never to fit a run; ADR-0027 records the addition and what the rules had said. 

12""" 

13 

14from dataclasses import dataclass 

15from enum import Enum 

16 

17from emblema.pretraining.domain.saturation.saturation_curve import SaturationCurve 

18 

19# Relative validation loss at the largest share at or past which nothing carried over: the 

20# channel mean's own error, so the held-out side was not learnt and the curve reads no data. 

21NOT_LEARNT_FROM = 1.0 

22# Relative validation loss this many times the relative training loss at the largest share: the 

23# model fits training windows it will not see again. The same ratio the masked-reconstruction 

24# assessment warns at, each side read against the trivial predictor's error on that side. 

25GENERALISATION_GAP = 1.5 

26# The last step up in data lowered the validation loss by less than this share of it: the corpus 

27# is learnt at this budget, and a larger share of it teaches nothing a smaller one did not. The 

28# same tolerance the assessment reads a doubled budget against. 

29PLATEAU_GAIN = 0.05 

30 

31 

32class SaturationVerdict(Enum): 

33 """What the curve shows about the corpus at this model size and budget. 

34 

35 Attributes: 

36 DATA_LIMITED: The loss still falls with more data at the largest share: the corpus is 

37 suited to pretraining and more of it, or more like it, would help. 

38 SATURATED: The loss stopped falling before the largest share: the corpus is exhausted at 

39 this size and budget, and a larger model over the same data gains nothing — more 

40 kinds of data would. 

41 OVERFITTING: The run over the whole corpus fits its training windows and not the held-out 

42 ones: too little data for the model, so a smaller model or a larger corpus. 

43 NOT_LEARNT: The run over the whole corpus does no better than the channel mean on the 

44 held-out side: nothing carried over, and the curve says nothing about the data — the 

45 held-out side, or the loss it is scored by, is what to look at first. 

46 """ 

47 

48 DATA_LIMITED = "data-limited" 

49 SATURATED = "saturated" 

50 OVERFITTING = "overfitting" 

51 NOT_LEARNT = "not learnt" 

52 

53 

54@dataclass(frozen=True) 

55class SaturationJudgement: 

56 """The verdict on one curve, with the two numbers it rests on and what they mean. 

57 

58 Attributes: 

59 verdict: What the curve shows. 

60 gain: Relative fall of the validation loss over the last step up in data. 

61 generalisation_ratio: Relative validation over relative training loss at the largest 

62 share. 

63 reading: The verdict in a sentence, with what to do about it. 

64 """ 

65 

66 verdict: SaturationVerdict 

67 gain: float 

68 generalisation_ratio: float 

69 reading: str 

70 

71 

72def judge(curve: SaturationCurve) -> SaturationJudgement: 

73 """Read the curve by the rules above, in their order of precedence.""" 

74 before, largest = curve.points[-2], curve.largest 

75 gain = curve.gains()[-1] 

76 ratio = largest.generalisation_ratio 

77 step = f"from {before.fraction:.0%} to {largest.fraction:.0%} of the training units" 

78 if largest.relative_validation_loss >= NOT_LEARNT_FROM: 

79 return SaturationJudgement( 

80 SaturationVerdict.NOT_LEARNT, 

81 gain, 

82 ratio, 

83 f"at the whole share validation is {largest.relative_validation_loss:.2f}× the trivial " 

84 "predictor's: nothing carried over to the held-out side, so the curve reads nothing " 

85 "about the data — the held-out side, or the loss it is scored by, comes before the " 

86 "corpus is judged.", 

87 ) 

88 if ratio > GENERALISATION_GAP: 

89 return SaturationJudgement( 

90 SaturationVerdict.OVERFITTING, 

91 gain, 

92 ratio, 

93 f"at the whole share validation is {ratio:.2f}× training, each against its own " 

94 f"trivial predictor, past {GENERALISATION_GAP:g}×: the model fits windows it will " 

95 "not see again, so the data is too little for it — a smaller model, or more units.", 

96 ) 

97 if gain < PLATEAU_GAIN: 

98 return SaturationJudgement( 

99 SaturationVerdict.SATURATED, 

100 gain, 

101 ratio, 

102 f"the step {step} lowered validation by {gain:.1%}, under {PLATEAU_GAIN:.0%}: the " 

103 "corpus is learnt at this size and budget, and a larger model over it gains nothing " 

104 "— other corpora would.", 

105 ) 

106 return SaturationJudgement( 

107 SaturationVerdict.DATA_LIMITED, 

108 gain, 

109 ratio, 

110 f"the step {step} lowered validation by {gain:.1%}: the data, not the model, is the " 

111 "limit at this budget — the corpus suits pretraining and more of it would help.", 

112 )
