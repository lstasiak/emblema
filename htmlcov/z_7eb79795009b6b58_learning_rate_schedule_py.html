2from dataclasses import dataclass 

3 

4from emblema.pretraining.domain.exceptions import InvalidLearningRateScheduleError 

5 

6 

7@dataclass(frozen=True, kw_only=True) 

8class LearningRateSchedule: 

9 """How the learning rate moves over a run: a linear warmup, then a cosine decay to a floor. 

10 

11 A factor of the peak rate per optimiser step, so the peak stays the optimiser's setting and the 

12 shape the run's. The decay keeps a verdict from being read off a model still circling its 

13 minimum; the warmup keeps a fresh model's first steps from undoing themselves. The warmup 

14 reaches the peak on its last step, so no step runs at zero; the decay starts at the peak and 

15 reaches the floor on the run's last step, or takes its one step at the peak where the warmup 

16 left it only one. Steps past the run stay at the floor. A pure function of the step: whoever 

17 trains holds the state. 

18 

19 Invariants: at least one step in all; warmup steps are not negative and leave at least one step 

20 to decay, since a run that only warms up would never use the floor it states; the floor lies in 

21 ``[0, 1]``. A floor of one with no warmup is the constant rate. 

22 

23 Attributes: 

24 warmup_steps: Steps over which the factor climbs to one. 

25 total_steps: Steps of the whole run. 

26 final_fraction: Factor the decay ends at, as a fraction of the peak. 

27 """ 

28 

29 warmup_steps: int 

30 total_steps: int 

31 final_fraction: float 

32 

33 def __post_init__(self) -> None: 

34 if self.total_steps < 1: 

35 raise InvalidLearningRateScheduleError( 

36 f"total_steps must be positive, got {self.total_steps}" 

37 ) 

38 if not 0 <= self.warmup_steps < self.total_steps: 

39 raise InvalidLearningRateScheduleError( 

40 f"warmup_steps must lie in [0, total_steps) so that a step is left to decay, got " 

41 f"{self.warmup_steps} of {self.total_steps}" 

42 ) 

43 if not 0.0 <= self.final_fraction <= 1.0: 

44 raise InvalidLearningRateScheduleError( 

45 f"final_fraction must lie in [0, 1], got {self.final_fraction}" 

46 ) 

47 

48 def factor(self, step: int) -> float: 

49 """The factor of the peak rate the optimiser uses at ``step``, counted from zero. 

50 

51 Raises: 

52 InvalidLearningRateScheduleError: If ``step`` is negative. 

53 """ 

54 if step < 0: 

55 raise InvalidLearningRateScheduleError(f"step must not be negative, got {step}") 

56 if step < self.warmup_steps: 

57 return (step + 1) / self.warmup_steps 

58 decaying = self.total_steps - self.warmup_steps 

59 progress = min((step - self.warmup_steps) / max(decaying - 1, 1), 1.0) 

60 cosine = 0.5 * (1.0 + math.cos(math.pi * progress)) 

61 return self.final_fraction + (1.0 - self.final_fraction) * cosine
