45BUFFERED_TOKENS = 1 << 20

48class WindowBlockWriter: 

49 """Writes token windows into one block file, holding no more of them than a buffer at a time. 

50 

51 Token columns spill to scratch files beside the block as they fill and are copied into it at 

52 the end, so memory does not grow with the corpus; the scratch never goes to the system's 

53 temporary directory, which inside a container may be memory. Every window is checked in the 

54 form it reads back in, cast to the stored width, so a window that precision would spoil is 

55 refused here rather than by the run that reads it. As a context manager the writer leaves 

56 nothing behind on an error. 

57 """

108 def add(self, window: TokenWindow, *, unit: int, start: float, end: float) -> None: 

109 """Append one window, cut from ``[start, end)`` of the unit at index ``unit``. 

110 

111 Raises: 

112 BlockClosedError: If the block has been closed. 

113 UnstorableWindowError: If the window, stored at the block's precision, would no 

114 longer be a valid window. 

115 """ 

116 if self._closed: 

117 raise BlockClosedError("the block is closed") 

118 stored = self._stored(window) 

119 self._units.append(unit) 

120 self._starts.append(start) 

121 self._ends.append(end) 

122 self._offsets.append(self._offsets[-1] + len(window)) 

123 for column, values in zip(self._columns, stored, strict=True): 

124 column.extend(values) 

125 if self._columns[0].buffered >= BUFFERED_TOKENS: 

126 for column in self._columns: 

127 column.flush()

140 def _stored(self, window: TokenWindow) -> tuple[list[object], ...]: 

141 """The window's columns as the block will give them back, checked to still be a window. 

142 

143 Casting is monotone but not one-to-one: two tokens ordered by a difference the stored 

144 width cannot hold read back out of canonical order. 

145 """ 

146 rows = (window.channel_ids, window.values, window.times, window.gaps, window.timeless) 

147 channel_ids, values, times, gaps, timeless = ( 

148 np.asarray(row, dtype=column.dtype).tolist() 

149 for column, row in zip(self._columns, rows, strict=True) 

150 ) 

151 try: 

152 TokenWindow( 

153 channel_ids=tuple(channel_ids), 

154 values=tuple(values), 

155 times=tuple(times), 

156 gaps=tuple(gaps), 

157 timeless=tuple(timeless), 

158 ) 

159 except InvalidTokenWindowError as error: 

160 raise UnstorableWindowError( 

161 f"a window of {len(window)} tokens is not a window once stored at " 

162 f"{self._columns[1].dtype}: {error}" 

163 ) from error 

164 return channel_ids, values, times, gaps, timeless

166 def _assemble(self, token_sizes: dict[str, int]) -> None: 

167 by_window = { 

168 UNIT_COLUMN: np.asarray(self._units, dtype=UNIT_DTYPE), 

169 START_COLUMN: np.asarray(self._starts, dtype=EXTENT_DTYPE), 

170 END_COLUMN: np.asarray(self._ends, dtype=EXTENT_DTYPE), 

171 OFFSET_COLUMN: np.asarray(self._offsets, dtype=OFFSET_DTYPE), 

172 } 

173 placement = self._placement( 

174 [(name, column.dtype.str, column.nbytes) for name, column in by_window.items()] 

175 + [(column.name, column.dtype, token_sizes[column.name]) for column in self._columns] 

176 ) 

177 header = encode_header( 

178 { 

179 "format": FORMAT_NAME, 

180 "version": VERSION, 

181 "windows": len(self._units), 

182 "tokens": self._offsets[-1], 

183 "columns": { 

184 name: {"dtype": dtype, "offset": offset, "bytes": size} 

185 for name, dtype, offset, size in placement 

186 }, 

187 } 

188 ) 

189 if HEADER_OFFSET + len(header) > DATA_OFFSET: 

190 raise HeaderOverflowError( 

191 f"header of {len(header)} bytes does not fit before the columns" 

192 ) 

193 try: 

194 with self._destination.open("wb") as block: 

195 block.write(MAGIC) 

196 block.write(struct.pack("<Q", len(header))) 

197 block.write(header) 

198 for name, _, offset, _ in placement: 

199 block.write(b"\0" * (offset - block.tell())) 

200 if name in by_window: 

201 block.write(by_window[name].tobytes()) 

202 else: 

203 with Path(self._scratch.name, name).open("rb") as column: 

204 shutil.copyfileobj(column, block) 

205 except BaseException: 

206 self._destination.unlink(missing_ok=True) 

207 raise

210 def _placement(columns: Sequence[tuple[str, str, int]]) -> list[tuple[str, str, int, int]]: 

211 """Name, type, offset and size of every column, in the order they are laid down.""" 

212 placement = [] 

213 offset = DATA_OFFSET 

214 for name, dtype, size in columns: 

215 placement.append((name, dtype, offset, size)) 

216 offset = padded_to_alignment(offset + size) 

217 return placement
